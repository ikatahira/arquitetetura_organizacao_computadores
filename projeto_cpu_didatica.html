Missão 1 · Montagem

🔌 Monte o circuito

Separe os materiais e faça as ligações antes de carregar o primeiro código.

🧰 Materiais

  • 1 Arduino Uno R3 + cabo USB
  • 1 protoboard 830 pontos
  • 4 LEDs (de qualquer cor) + 4 resistores de 300 Ω
  • 1 LCD 16×2 (fase 3)
  • 1 resistor de 1 kΩ (contraste do LCD) e 1 de 300 Ω (luz de fundo do LCD)
  • Jumpers

⚠️ Antes de começar

  • O contraste do LCD é o problema mais comum: se aparecem só quadradinhos escuros, mexa no resistor do V0 (ou use um potenciômetro).
  • Os comandos são enviados como linha: no Monitor Serial, escolha “Nova linha” ao lado do campo de envio.
  • No Tinkercad, o LCD 16×2 já vem com os mesmos pinos.

Missão 2 · Fase 1

📥 Memória, PC e busca (fetch)

Construir a primeira parte do processador: a memória de 16 bytes, o contador de programa (PC) e a busca da instrução. Os LEDs mostram o PC em binário.

💻 Código da fase 1 cpu1_memoria_fetch.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  CPU DIDATICA - FASE 1: memoria, contador de programa (PC) e busca (fetch)

  Vamos construir um processador minusculo, no estilo SAP-1:
    - memoria unica de 16 bytes (programa E dados juntos: arquitetura Von Neumann)
    - PC (Program Counter): endereco da proxima instrucao (4 bits)
    - IR (Instruction Register): instrucao que esta sendo executada (8 bits)
    - Instrucao de 8 bits = 4 bits de OPCODE + 4 bits de OPERANDO

  Nesta fase so fazemos a BUSCA (fetch):  IR <- memoria[PC] ; PC <- PC + 1
  Os 4 LEDs mostram o PC em binario.

  Comandos no Monitor Serial:  s = um passo | r = reset | m = mostrar memoria

  Ligacoes: 4 LEDs (cada um com 300 ohms) nos pinos 10, 9, 8, 7 (bit 0 ... bit 3)
*/
const uint8_t LEDS[4] = {10, 9, 8, 7};

// Programa de exemplo ("contagem regressiva"), em hexadecimal:
uint8_t memoria[16] = {
  0x15, 0x80, 0x61, 0xB1, 0xF0, 0x00, 0x00, 0x00,
  0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
};
uint8_t PC = 0;
uint8_t IR = 0;

void imprimirBinario(uint8_t v, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((v >> i) & 1);
}

void mostrarPC() {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (PC >> b) & 1);
}

void mostrarMemoria() {
  Serial.println(F("end | conteudo (HEX) | binario"));
  for (uint8_t i = 0; i < 16; i++) {
    Serial.print(i, HEX); Serial.print(F("   | 0x"));
    if (memoria[i] < 16) Serial.print('0');
    Serial.print(memoria[i], HEX); Serial.print(F("           | "));
    imprimirBinario(memoria[i], 8);
    if (i == PC) Serial.print(F("   <-- PC"));
    Serial.println();
  }
}

void passoBusca() {
  IR = memoria[PC];                      // T1: IR <- mem[PC]
  Serial.print(F("PC = ")); Serial.print(PC, HEX);
  Serial.print(F(" | IR = 0x")); if (IR < 16) Serial.print('0'); Serial.print(IR, HEX);
  Serial.print(F(" = ")); imprimirBinario(IR, 8);
  Serial.print(F(" | opcode = ")); imprimirBinario(IR >> 4, 4);
  Serial.print(F(" | operando = ")); imprimirBinario(IR & 0xF, 4);
  Serial.println();
  PC = (PC + 1) & 0xF;                   // T2: PC <- PC + 1 (so 4 bits: volta para 0 depois de F)
  mostrarPC();
}

void setup() {
  Serial.begin(9600);
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  mostrarPC();
  Serial.println(F("=== CPU DIDATICA - Fase 1 (busca) ==="));
  Serial.println(F("Comandos: s = passo | r = reset | m = memoria"));
  mostrarMemoria();
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 's') passoBusca();
    if (c == 'r') { PC = 0; IR = 0; mostrarPC(); Serial.println(F(">> reset")); }
    if (c == 'm') mostrarMemoria();
  }
}
Missão 3 · Fase 2

⚙️ Decodificação, execução e o conjunto de instruções

Completar o ciclo: decodificar o opcode e executar 12 instruções (LDI, LDA, STA, ADD, ADDM, SUB, AND, OUT, JMP, JZ, JNZ, XOR, SUBM, HLT). Rodar programas de exemplo e escrever o seu.

💻 Código da fase 2 cpu2_decodifica_executa.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  CPU DIDATICA - FASE 2: decodificacao, execucao e conjunto de instrucoes

  Ciclo de instrucao (busca -> decodificacao -> execucao):
    T1: IR <- mem[PC] ; PC <- PC + 1       (busca)
    T2: a unidade de controle olha o OPCODE (decodifica)
    T3: executa e atualiza A, a flag Z e/ou a saida

  Registradores: A (acumulador, 8 bits), PC (4 bits), IR (8 bits), flag Z (zero)
  Conjunto de instrucoes (opcode em hexadecimal):
    0 NOP            1 LDI n  (A <- n)          imediato
    2 LDA m  (A <- mem[m])   direto             3 STA m  (mem[m] <- A)
    4 ADD n  (A <- A + n)    imediato           5 ADDM m (A <- A + mem[m]) direto
    6 SUB n  (A <- A - n)    imediato           7 AND n
    8 OUT    (saida <- A)                       9 JMP n  (PC <- n)
    A JZ n   (salta se Z=1)  B JNZ n (salta se Z=0)
    C XOR n                  D SUBM m (A <- A - mem[m])      F HLT (para)

  Comandos no Monitor Serial:
    s = um passo | g = executar ate HLT (maximo 60 passos) | r = reset
    m = mostrar memoria | l0, l1, l2 = carregar programa de exemplo
    w <end> <hex> = escrever na memoria. Exemplo:  w 3 2F

  Ligacoes: 4 LEDs (300 ohms) nos pinos 10, 9, 8, 7 mostram os 4 bits menos
  significativos da SAIDA.
*/
const uint8_t LEDS[4] = {10, 9, 8, 7};

uint8_t memoria[16];
uint8_t PC = 0, IR = 0, A = 0, saida = 0;
bool Z = false, parada = false;
unsigned long passos = 0;

const char* NOMES[16] = {"NOP", "LDI", "LDA", "STA", "ADD", "ADDM", "SUB", "AND",
                         "OUT", "JMP", "JZ", "JNZ", "XOR", "SUBM", "???", "HLT"};

void carregarPrograma(uint8_t n) {
  for (uint8_t i = 0; i < 16; i++) memoria[i] = 0;
  if (n == 0) {              // contagem regressiva 5,4,3,2,1
    const uint8_t p[] = {0x15, 0x80, 0x61, 0xB1, 0xF0};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
  } else if (n == 1) {       // soma de duas variaveis na memoria: 3 + 4
    const uint8_t p[] = {0x2E, 0x5F, 0x3D, 0x80, 0xF0};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
    memoria[14] = 3; memoria[15] = 4;
  } else {                   // Fibonacci (1 1 2 3 5 8 13 ...)
    const uint8_t p[] = {0x2F, 0x80, 0x5E, 0x3D, 0x2F, 0x3E, 0x2D, 0x3F, 0x90};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
    memoria[14] = 0; memoria[15] = 1;
  }
}

void resetar() { PC = 0; IR = 0; A = 0; saida = 0; Z = false; parada = false; passos = 0; }

void mostrarSaida() {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (saida >> b) & 1);
}

void mostrarMemoria() {
  for (uint8_t i = 0; i < 16; i++) {
    Serial.print(i, HEX); Serial.print(F(": "));
    if (memoria[i] < 16) Serial.print('0');
    Serial.print(memoria[i], HEX);
    Serial.print(F("  "));
    Serial.print(NOMES[memoria[i] >> 4]); Serial.print(' '); Serial.print(memoria[i] & 0xF, HEX);
    if (i == PC) Serial.print(F("   <-- PC"));
    Serial.println();
  }
}

void atualizaZ() { Z = (A == 0); }

void passo() {
  if (parada) { Serial.println(F("CPU parada (HLT). Use r para reiniciar.")); return; }

  // T1: busca
  uint8_t pcAntes = PC;
  IR = memoria[PC];
  PC = (PC + 1) & 0xF;
  // T2: decodificacao
  uint8_t op = IR >> 4, n = IR & 0xF;
  // T3: execucao
  switch (op) {
    case 0x0: break;
    case 0x1: A = n; atualizaZ(); break;
    case 0x2: A = memoria[n]; atualizaZ(); break;
    case 0x3: memoria[n] = A; break;
    case 0x4: A = A + n; atualizaZ(); break;
    case 0x5: A = A + memoria[n]; atualizaZ(); break;
    case 0x6: A = A - n; atualizaZ(); break;
    case 0x7: A = A & n; atualizaZ(); break;
    case 0x8: saida = A; mostrarSaida(); break;
    case 0x9: PC = n; break;
    case 0xA: if (Z) PC = n; break;
    case 0xB: if (!Z) PC = n; break;
    case 0xC: A = A ^ n; atualizaZ(); break;
    case 0xD: A = A - memoria[n]; atualizaZ(); break;
    case 0xF: parada = true; break;
    default: break;
  }
  passos++;

  Serial.print(F("PC=")); Serial.print(pcAntes, HEX);
  Serial.print(F(" IR=0x")); if (IR < 16) Serial.print('0'); Serial.print(IR, HEX);
  Serial.print(F(" ")); Serial.print(NOMES[op]); Serial.print(' '); Serial.print(n, HEX);
  Serial.print(F(" | A=")); Serial.print(A);
  Serial.print(F(" Z=")); Serial.print(Z);
  Serial.print(F(" SAIDA=")); Serial.println(saida);
}

// Le uma linha do Serial e executa comandos de varios caracteres
void tratarLinha(String l) {
  l.trim();
  if (l.length() == 0) return;
  char c = l.charAt(0);
  if (c == 's') passo();
  else if (c == 'r') { resetar(); mostrarSaida(); Serial.println(F(">> reset")); }
  else if (c == 'm') mostrarMemoria();
  else if (c == 'g') {
    for (uint8_t i = 0; i < 60 && !parada; i++) { passo(); delay(150); }
  }
  else if (c == 'l') { carregarPrograma(l.charAt(1) - '0'); resetar(); mostrarSaida(); Serial.println(F(">> programa carregado")); mostrarMemoria(); }
  else if (c == 'w') {
    int sp = l.indexOf(' ');
    int sp2 = l.indexOf(' ', sp + 1);
    if (sp > 0 && sp2 > 0) {
      long end = strtol(l.substring(sp + 1, sp2).c_str(), NULL, 16);
      long val = strtol(l.substring(sp2 + 1).c_str(), NULL, 16);
      if (end >= 0 && end < 16) { memoria[end] = val & 0xFF; Serial.println(F(">> escrito")); }
    }
  }
}

void setup() {
  Serial.begin(9600);
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  carregarPrograma(0);
  resetar();
  mostrarSaida();
  Serial.println(F("=== CPU DIDATICA - Fase 2 ==="));
  Serial.println(F("Comandos: s passo | g executa | r reset | m memoria | l0 l1 l2 programas | w end hex"));
  mostrarMemoria();
}

void loop() {
  if (Serial.available()) {
    String l = Serial.readStringUntil('\n');
    tratarLinha(l);
  }
}
Missão 4 · Fase 3

📊 Painel LCD, ciclos e desempenho

Acrescentar o LCD como painel do processador e medir o desempenho: ciclos gastos por instrução, CPI e tempo total para um clock de 1 a 20 Hz.

💻 Código da fase 3 cpu3_painel_lcd.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  CPU DIDATICA - FASE 3: painel LCD, ciclos de clock e desempenho

  Igual a Fase 2, com tres novidades:
    1. Um LCD 16x2 funciona como "painel de controle" (PC, A, SAIDA, flag Z).
    2. Cada instrucao gasta CICLOS: busca = 1, execucao = 1 (2 se acessa a memoria).
       O Monitor Serial mostra o total de ciclos, o CPI (ciclos por instrucao)
       e o tempo total para um clock simulado de FREQ_HZ.
    3. Comando 'f <hz>' muda a frequencia do clock simulado (1 a 20 Hz).

  Comandos: s passo | g executa | r reset | l0 l1 l2 programas | f <hz> | m memoria

  Ligacoes do LCD 16x2 (modo 4 bits):
    LCD RS -> 12   E -> 11   D4 -> 5   D5 -> 4   D6 -> 3   D7 -> 2
    LCD VSS, RW, K(-) -> GND   VDD, A(+) -> 5V (A com resistor de 300 ohms)
    V0 (contraste) -> GND por um resistor de 1 kohm (ou potenciometro de 10 kohm)
  LEDs de saida (300 ohms) nos pinos 10, 9, 8, 7.
*/
#include <LiquidCrystal.h>
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

const uint8_t LEDS[4] = {10, 9, 8, 7};
uint8_t memoria[16];
uint8_t PC = 0, IR = 0, A = 0, saida = 0;
bool Z = false, parada = false;
unsigned long instrucoes = 0, ciclos = 0;
uint8_t freqHz = 4;

const char* NOMES[16] = {"NOP", "LDI", "LDA", "STA", "ADD", "ADDM", "SUB", "AND",
                         "OUT", "JMP", "JZ", "JNZ", "XOR", "SUBM", "???", "HLT"};

void carregarPrograma(uint8_t n) {
  for (uint8_t i = 0; i < 16; i++) memoria[i] = 0;
  if (n == 0) {
    const uint8_t p[] = {0x15, 0x80, 0x61, 0xB1, 0xF0};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
  } else if (n == 1) {
    const uint8_t p[] = {0x2E, 0x5F, 0x3D, 0x80, 0xF0};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
    memoria[14] = 3; memoria[15] = 4;
  } else {
    const uint8_t p[] = {0x2F, 0x80, 0x5E, 0x3D, 0x2F, 0x3E, 0x2D, 0x3F, 0x90};
    for (uint8_t i = 0; i < sizeof(p); i++) memoria[i] = p[i];
    memoria[14] = 0; memoria[15] = 1;
  }
}

void resetar() { PC = 0; IR = 0; A = 0; saida = 0; Z = false; parada = false; instrucoes = 0; ciclos = 0; }

void mostrarSaida() {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (saida >> b) & 1);
}

void atualizarLcd(const char* mnem, uint8_t n) {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("PC:"); lcd.print(PC, HEX);
  lcd.print(' '); lcd.print(mnem); lcd.print(' '); lcd.print(n, HEX);
  lcd.setCursor(0, 1);
  lcd.print("A:"); lcd.print(A);
  lcd.print(" Z:"); lcd.print(Z ? 1 : 0);
  lcd.print(" S:"); lcd.print(saida);
}

void atualizaZ() { Z = (A == 0); }

void passo() {
  if (parada) { Serial.println(F("CPU parada (HLT). Use r para reiniciar.")); return; }
  IR = memoria[PC];
  PC = (PC + 1) & 0xF;
  uint8_t op = IR >> 4, n = IR & 0xF;
  uint8_t custo = 2;                       // 1 ciclo de busca + 1 de execucao
  switch (op) {
    case 0x0: break;
    case 0x1: A = n; atualizaZ(); break;
    case 0x2: A = memoria[n]; atualizaZ(); custo = 3; break;
    case 0x3: memoria[n] = A; custo = 3; break;
    case 0x4: A = A + n; atualizaZ(); break;
    case 0x5: A = A + memoria[n]; atualizaZ(); custo = 3; break;
    case 0x6: A = A - n; atualizaZ(); break;
    case 0x7: A = A & n; atualizaZ(); break;
    case 0x8: saida = A; mostrarSaida(); break;
    case 0x9: PC = n; break;
    case 0xA: if (Z) PC = n; break;
    case 0xB: if (!Z) PC = n; break;
    case 0xC: A = A ^ n; atualizaZ(); break;
    case 0xD: A = A - memoria[n]; atualizaZ(); custo = 3; break;
    case 0xF: parada = true; break;
    default: break;
  }
  instrucoes++;
  ciclos += custo;
  atualizarLcd(NOMES[op], n);

  Serial.print(NOMES[op]); Serial.print(' '); Serial.print(n, HEX);
  Serial.print(F(" | A=")); Serial.print(A);
  Serial.print(F(" SAIDA=")); Serial.print(saida);
  Serial.print(F(" | ciclos desta instrucao = ")); Serial.println(custo);
}

void relatorioDesempenho() {
  if (instrucoes == 0) return;
  float cpi = (float)ciclos / instrucoes;
  float tempo = (float)ciclos / freqHz;
  Serial.println(F("--- DESEMPENHO ---"));
  Serial.print(F("instrucoes = ")); Serial.println(instrucoes);
  Serial.print(F("ciclos     = ")); Serial.println(ciclos);
  Serial.print(F("CPI (ciclos por instrucao) = ")); Serial.println(cpi, 2);
  Serial.print(F("clock simulado = ")); Serial.print(freqHz); Serial.println(F(" Hz"));
  Serial.print(F("tempo = ciclos / clock = ")); Serial.print(tempo, 2); Serial.println(F(" s"));
}

void executarTudo() {
  for (uint8_t i = 0; i < 60 && !parada; i++) {
    passo();
    delay(1000UL / freqHz);                 // o "clock" da nossa CPU
  }
  relatorioDesempenho();
}

void tratarLinha(String l) {
  l.trim();
  if (l.length() == 0) return;
  char c = l.charAt(0);
  if (c == 's') passo();
  else if (c == 'g') executarTudo();
  else if (c == 'r') { resetar(); mostrarSaida(); lcd.clear(); lcd.print("CPU pronta"); Serial.println(F(">> reset")); }
  else if (c == 'l') { carregarPrograma(l.charAt(1) - '0'); resetar(); mostrarSaida(); Serial.println(F(">> programa carregado")); }
  else if (c == 'f') { int v = l.substring(1).toInt(); if (v >= 1 && v <= 20) { freqHz = v; Serial.print(F(">> clock = ")); Serial.print(freqHz); Serial.println(F(" Hz")); } }
  else if (c == 'm') {
    for (uint8_t i = 0; i < 16; i++) {
      Serial.print(i, HEX); Serial.print(F(": "));
      if (memoria[i] < 16) Serial.print('0');
      Serial.println(memoria[i], HEX);
    }
  }
}

void setup() {
  Serial.begin(9600);
  lcd.begin(16, 2);
  lcd.print("CPU Didatica");
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  carregarPrograma(0);
  resetar();
  mostrarSaida();
  Serial.println(F("=== CPU DIDATICA - Fase 3 ==="));
  Serial.println(F("Comandos: s | g | r | l0 l1 l2 | f <hz> | m"));
}

void loop() {
  if (Serial.available()) {
    String l = Serial.readStringUntil('\n');
    tratarLinha(l);
  }
}
