🔐 Cofre RFID com arquivo na EEPROM

Um leitor RFID abre uma trava (servo) para cartões cadastrados. Os cartões ficam em uma tabela na EEPROM, com permissões, checksum e registro de acessos, como um pequeno sistema de arquivos.

Missão 1 · Montagem

🔌 Monte o circuito

Separe os materiais e faça as ligações antes de carregar o primeiro código.

🧰 Materiais

  • 1 Arduino Uno R3 + cabo USB
  • 1 protoboard 830 pontos
  • 1 kit RFID MFRC522 (leitor + cartão + chaveiro)
  • 1 conversor de nível lógico (4 canais)
  • 1 micro servo SG90 (fase 3) + cabo extensor, se precisar
  • 1 buzzer passivo
  • 1 LED verde e 1 LED vermelho + 2 resistores de 300 Ω
  • Jumpers macho-macho e macho-fêmea

⚠️ Antes de começar

  • O MFRC522 trabalha com 3,3 V. Ligar o módulo no 5V pode queimá-lo.
  • Só funciona na placa física: o Tinkercad não tem o MFRC522. Para rodar a fase 1 no simulador, troque o leitor por entradas manuais.
  • Se o servo reiniciar a placa, alimente-o por outra fonte com o GND em comum.
  • Guarde os cartões: o UID é único e é a “chave” do sistema. Não é um sistema de segurança real (o UID pode ser clonado).

Missão 2 · Fase 1

📶 SPI, níveis lógicos e UID do cartão

Ligar o leitor pelo barramento SPI com conversor de nível e ler o UID: em hexadecimal, binário e decimal, mais um checksum XOR.

💻 Código da fase 1 cofre1_leitor_uid.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  COFRE RFID - FASE 1: barramento SPI, niveis logicos 5 V x 3,3 V e o UID do cartao

  O leitor RFID MFRC522 conversa com o Arduino pelo barramento SPI (4 fios + reset).
  ATENCAO: o MFRC522 funciona com 3,3 V. O Uno trabalha com 5 V.
    - Alimente o modulo com o pino 3.3V do Uno (NUNCA com 5V).
    - Use o CONVERSOR DE NIVEL LOGICO (lado HV = 5 V, lado LV = 3,3 V) nos sinais
      que saem do Arduino: SCK, MOSI, SDA(SS) e RST.
    - MISO (saida do modulo, 3,3 V) pode ir direto ao Arduino: 3,3 V ja e lido como ALTO.

  Ligacoes (Uno -> conversor -> MFRC522):
    13 (SCK)  -> HV1 / LV1 -> SCK       11 (MOSI) -> HV2 / LV2 -> MOSI
    10 (SS)   -> HV3 / LV3 -> SDA       9  (RST)  -> HV4 / LV4 -> RST
    12 (MISO) <- MISO (direto)          3.3V -> 3.3V do modulo e do lado LV
    5V -> lado HV do conversor          GND -> GND de tudo
    LED verde (300 ohms) no pino 7 | LED vermelho (300 ohms) no pino 6 | buzzer passivo no pino 5
  Biblioteca: MFRC522 (de Miguel Balboa). Instale pelo Gerenciador de Bibliotecas da IDE.

  O que o programa faz: ao aproximar um cartao ou chaveiro, mostra o UID
  (identificador unico, 4 bytes) em hexadecimal e binario, mais o tipo do cartao.
*/
#include <SPI.h>
#include <MFRC522.h>

const uint8_t PIN_SS = 10, PIN_RST = 9;
const uint8_t PIN_VERDE = 7, PIN_VERMELHO = 6, PIN_BUZZER = 5;
MFRC522 leitor(PIN_SS, PIN_RST);

void imprimirBinario(uint8_t v) {
  for (int8_t i = 7; i >= 0; i--) Serial.print((v >> i) & 1);
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_VERDE, OUTPUT); pinMode(PIN_VERMELHO, OUTPUT); pinMode(PIN_BUZZER, OUTPUT);
  SPI.begin();
  leitor.PCD_Init();
  Serial.println(F("=== COFRE RFID - Fase 1: leitor de UID ==="));
  Serial.print(F("Versao do chip MFRC522 (0x92 = v2.0, 0x91 = v1.0): 0x"));
  Serial.println(leitor.PCD_ReadRegister(MFRC522::VersionReg), HEX);
  Serial.println(F("Aproxime um cartao..."));
}

void loop() {
  if (!leitor.PICC_IsNewCardPresent() || !leitor.PICC_ReadCardSerial()) return;

  Serial.print(F("Tipo: ")); Serial.println(leitor.PICC_GetTypeName(leitor.PICC_GetType(leitor.uid.sak)));
  Serial.print(F("UID (")); Serial.print(leitor.uid.size); Serial.println(F(" bytes):"));
  uint8_t xorTotal = 0;
  for (uint8_t i = 0; i < leitor.uid.size; i++) {
    Serial.print(F("  byte ")); Serial.print(i);
    Serial.print(F(": HEX 0x")); if (leitor.uid.uidByte[i] < 16) Serial.print('0');
    Serial.print(leitor.uid.uidByte[i], HEX);
    Serial.print(F(" | BIN ")); imprimirBinario(leitor.uid.uidByte[i]);
    Serial.print(F(" | DEC ")); Serial.println(leitor.uid.uidByte[i]);
    xorTotal ^= leitor.uid.uidByte[i];
  }
  Serial.print(F("XOR dos bytes (checksum simples): 0x")); Serial.println(xorTotal, HEX);

  digitalWrite(PIN_VERDE, HIGH);
  tone(PIN_BUZZER, 1500, 120);
  delay(400);
  digitalWrite(PIN_VERDE, LOW);
  leitor.PICC_HaltA();
}
Missão 3 · Fase 2

🗄️ Tabela de cartões na EEPROM

Criar um “sistema de arquivos” minúsculo: cabeçalho com número mágico, tabela de 16 registros (UID, permissões, acessos, checksum) e comandos para listar, cadastrar, remover e formatar.

💻 Código da fase 2 cofre2_cartoes_eeprom.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  COFRE RFID - FASE 2: um mini "sistema de arquivos" na EEPROM

  Os cartoes autorizados ficam guardados na EEPROM (memoria que nao se apaga).
  Organizacao (como um disco minusculo):
      endereco 0..3  : CABECALHO  = numero magico 0xC0, versao, quantidade, reservado
      endereco 4...  : TABELA de ate 16 REGISTROS de 8 bytes cada:
           uid[4] | permissoes (1 byte) | acessos (2 bytes) | checksum (1 byte)
  O checksum (XOR dos 7 primeiros bytes) detecta registro corrompido.
  Permissoes: bit 0 = pode abrir | bit 1 = administrador

  Comandos no Monitor Serial:
     l = listar cartoes | a = cadastrar (o proximo cartao lido) | r = remover (o proximo cartao)
     f = formatar a tabela (apaga tudo)

  Ligacoes: as mesmas da Fase 1.
*/
#include <SPI.h>
#include <MFRC522.h>
#include <EEPROM.h>

const uint8_t PIN_SS = 10, PIN_RST = 9;
const uint8_t PIN_VERDE = 7, PIN_VERMELHO = 6, PIN_BUZZER = 5;
MFRC522 leitor(PIN_SS, PIN_RST);

const uint8_t MAGICO = 0xC0;
const uint8_t MAX_CARTOES = 16;
const uint8_t TAM_REG = 8;
const int INICIO_TABELA = 4;

struct Cartao {
  uint8_t  uid[4];
  uint8_t  perm;
  uint16_t acessos;
  uint8_t  soma;
};

char modo = 'u';          // u = uso normal | a = cadastrar | r = remover

uint8_t calcularSoma(const Cartao &c) {
  const uint8_t *p = (const uint8_t *)&c;
  uint8_t x = 0;
  for (uint8_t i = 0; i < TAM_REG - 1; i++) x ^= p[i];
  return x;
}

int enderecoDe(uint8_t i) { return INICIO_TABELA + i * TAM_REG; }

void lerRegistro(uint8_t i, Cartao &c) { EEPROM.get(enderecoDe(i), c); }
void gravarRegistro(uint8_t i, Cartao &c) { c.soma = calcularSoma(c); EEPROM.put(enderecoDe(i), c); }

void formatar() {
  EEPROM.update(0, MAGICO); EEPROM.update(1, 1); EEPROM.update(2, 0); EEPROM.update(3, 0);
  Cartao vazio;
  memset(&vazio, 0xFF, sizeof(vazio));
  for (uint8_t i = 0; i < MAX_CARTOES; i++) EEPROM.put(enderecoDe(i), vazio);
  Serial.println(F(">> tabela formatada"));
}

bool registroLivre(const Cartao &c) { return c.uid[0] == 0xFF && c.uid[1] == 0xFF && c.uid[2] == 0xFF && c.uid[3] == 0xFF; }

// Busca linear: devolve o indice do cartao ou -1
int buscar(const uint8_t *uid) {
  Cartao c;
  for (uint8_t i = 0; i < MAX_CARTOES; i++) {
    lerRegistro(i, c);
    if (!registroLivre(c) && memcmp(c.uid, uid, 4) == 0) return i;
  }
  return -1;
}

void listar() {
  Serial.println(F("--- tabela de cartoes (EEPROM) ---"));
  Cartao c;
  uint8_t n = 0;
  for (uint8_t i = 0; i < MAX_CARTOES; i++) {
    lerRegistro(i, c);
    if (registroLivre(c)) continue;
    n++;
    Serial.print(i); Serial.print(F(": UID "));
    for (uint8_t k = 0; k < 4; k++) { if (c.uid[k] < 16) Serial.print('0'); Serial.print(c.uid[k], HEX); Serial.print(' '); }
    Serial.print(F("| perm=")); Serial.print(c.perm, BIN);
    Serial.print(F(" | acessos=")); Serial.print(c.acessos);
    Serial.print(F(" | checksum ")); Serial.println(c.soma == calcularSoma(c) ? F("ok") : F("CORROMPIDO"));
  }
  Serial.print(F("Total: ")); Serial.print(n); Serial.print(F(" de ")); Serial.println(MAX_CARTOES);
}

void sinal(bool bom) {
  digitalWrite(bom ? PIN_VERDE : PIN_VERMELHO, HIGH);
  tone(PIN_BUZZER, bom ? 1500 : 300, bom ? 120 : 400);
  delay(500);
  digitalWrite(PIN_VERDE, LOW); digitalWrite(PIN_VERMELHO, LOW);
}

void tratarCartao(const uint8_t *uid) {
  int idx = buscar(uid);
  Cartao c;
  if (modo == 'a') {
    if (idx >= 0) { Serial.println(F("Cartao ja cadastrado.")); sinal(false); }
    else {
      for (uint8_t i = 0; i < MAX_CARTOES; i++) {
        lerRegistro(i, c);
        if (registroLivre(c)) {
          memcpy(c.uid, uid, 4); c.perm = 0b01; c.acessos = 0;
          gravarRegistro(i, c);
          Serial.print(F("Cadastrado na posicao ")); Serial.println(i);
          sinal(true);
          modo = 'u';
          return;
        }
      }
      Serial.println(F("Tabela cheia!")); sinal(false);
    }
    modo = 'u';
  } else if (modo == 'r') {
    if (idx < 0) { Serial.println(F("Cartao nao encontrado.")); sinal(false); }
    else {
      memset(&c, 0xFF, sizeof(c));
      EEPROM.put(enderecoDe(idx), c);
      Serial.println(F("Cartao removido.")); sinal(true);
    }
    modo = 'u';
  } else {
    if (idx >= 0) {
      lerRegistro(idx, c);
      c.acessos++;
      gravarRegistro(idx, c);             // so grava quando o contador muda
      Serial.print(F("ACESSO PERMITIDO | acessos deste cartao: ")); Serial.println(c.acessos);
      sinal(true);
    } else {
      Serial.println(F("ACESSO NEGADO"));
      sinal(false);
    }
  }
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_VERDE, OUTPUT); pinMode(PIN_VERMELHO, OUTPUT); pinMode(PIN_BUZZER, OUTPUT);
  SPI.begin();
  leitor.PCD_Init();
  if (EEPROM.read(0) != MAGICO) { Serial.println(F("EEPROM sem formato: formatando...")); formatar(); }
  Serial.println(F("=== COFRE RFID - Fase 2 ==="));
  Serial.println(F("Comandos: l listar | a cadastrar | r remover | f formatar"));
  Serial.print(F("sizeof(Cartao) = ")); Serial.print(sizeof(Cartao)); Serial.println(F(" bytes"));
  listar();
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'l') listar();
    if (c == 'a') { modo = 'a'; Serial.println(F(">> aproxime o cartao a CADASTRAR")); }
    if (c == 'r') { modo = 'r'; Serial.println(F(">> aproxime o cartao a REMOVER")); }
    if (c == 'f') formatar();
  }
  if (!leitor.PICC_IsNewCardPresent() || !leitor.PICC_ReadCardSerial()) return;
  tratarCartao(leitor.uid.uidByte);
  leitor.PICC_HaltA();
}
Missão 4 · Fase 3

🔑 Trava com servo, administrador e log circular

Fechar o projeto: o servo abre a trava por 3 s, um cartão administrador cadastra novos cartões sem computador e um log circular de 16 eventos registra os acessos.

💻 Código da fase 3 cofre3_servo_permissoes_log.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  COFRE RFID - FASE 3: trava com servo, permissoes (administrador) e registro de acessos

  Novidades:
    - Um micro servo SG90 faz de TRAVA: 0 graus = trancado, 90 graus = aberto (3 s).
    - Cadastro SEM computador: passe o cartao ADMINISTRADOR e, em ate 5 segundos,
      passe o cartao NOVO. Ele entra na tabela com permissao de abrir.
    - Um LOG circular de 16 eventos na EEPROM (a partir do endereco 160):
        evento = [indice do cartao | tipo | segundos desde o boot, 2 bytes]
      Quando enche, o evento novo sobrescreve o mais antigo.
    - Comandos: l listar cartoes | g mostrar o log | f formatar tudo | m promover o
      PRIMEIRO cartao cadastrado a administrador (use na primeira vez)
      e 'a' cadastra o proximo cartao lido (como na Fase 2)

  Ligacoes: as da Fase 1 e mais:  servo SG90: fio marrom GND, vermelho 5V, laranja pino 3
  (se o servo reiniciar a placa, alimente-o por uma fonte separada com GND comum).
*/
#include <SPI.h>
#include <MFRC522.h>
#include <EEPROM.h>
#include <Servo.h>

const uint8_t PIN_SS = 10, PIN_RST = 9;
const uint8_t PIN_VERDE = 7, PIN_VERMELHO = 6, PIN_BUZZER = 5, PIN_SERVO = 3;
MFRC522 leitor(PIN_SS, PIN_RST);
Servo trava;

const uint8_t MAGICO = 0xC0;
const uint8_t MAX_CARTOES = 16;
const uint8_t TAM_REG = 8;
const int INICIO_TABELA = 4;
const int INICIO_LOG = 160;
const uint8_t MAX_LOG = 16;
const uint8_t PERM_ABRIR = 0b01, PERM_ADMIN = 0b10;
const unsigned long JANELA_ADMIN_MS = 5000;

struct Cartao { uint8_t uid[4]; uint8_t perm; uint16_t acessos; uint8_t soma; };
struct Evento { uint8_t cartao; uint8_t tipo; uint16_t segundos; };

enum { EV_ABRIU = 1, EV_NEGADO = 2, EV_CADASTRO = 3 };
char modo = 'u';
unsigned long ate = 0;             // quando termina a janela do administrador
uint8_t ponteiroLog = 0;

uint8_t calcularSoma(const Cartao &c) {
  const uint8_t *p = (const uint8_t *)&c;
  uint8_t x = 0;
  for (uint8_t i = 0; i < TAM_REG - 1; i++) x ^= p[i];
  return x;
}
int enderecoDe(uint8_t i) { return INICIO_TABELA + i * TAM_REG; }
void lerRegistro(uint8_t i, Cartao &c) { EEPROM.get(enderecoDe(i), c); }
void gravarRegistro(uint8_t i, Cartao &c) { c.soma = calcularSoma(c); EEPROM.put(enderecoDe(i), c); }
bool registroLivre(const Cartao &c) { return c.uid[0] == 0xFF && c.uid[1] == 0xFF && c.uid[2] == 0xFF && c.uid[3] == 0xFF; }

void formatar() {
  EEPROM.update(0, MAGICO); EEPROM.update(1, 1); EEPROM.update(2, 0); EEPROM.update(3, 0);
  Cartao vazio; memset(&vazio, 0xFF, sizeof(vazio));
  for (uint8_t i = 0; i < MAX_CARTOES; i++) EEPROM.put(enderecoDe(i), vazio);
  Evento e; memset(&e, 0xFF, sizeof(e));
  for (uint8_t i = 0; i < MAX_LOG; i++) EEPROM.put(INICIO_LOG + i * sizeof(Evento), e);
  Serial.println(F(">> tudo formatado"));
}

int buscar(const uint8_t *uid) {
  Cartao c;
  for (uint8_t i = 0; i < MAX_CARTOES; i++) {
    lerRegistro(i, c);
    if (!registroLivre(c) && memcmp(c.uid, uid, 4) == 0) return i;
  }
  return -1;
}

void registrarEvento(uint8_t cartao, uint8_t tipo) {
  Evento e = { cartao, tipo, (uint16_t)(millis() / 1000) };
  EEPROM.put(INICIO_LOG + ponteiroLog * sizeof(Evento), e);
  ponteiroLog = (ponteiroLog + 1) % MAX_LOG;     // buffer circular
}

void descobrirPonteiroLog() {                    // acha a proxima posicao livre (0xFF) do log
  Evento e;
  ponteiroLog = 0;
  for (uint8_t i = 0; i < MAX_LOG; i++) {
    EEPROM.get(INICIO_LOG + i * sizeof(Evento), e);
    if (e.tipo == 0xFF) { ponteiroLog = i; return; }
  }
}

void mostrarLog() {
  Serial.println(F("--- log de acessos (posicao | cartao | evento | segundos desde o boot) ---"));
  Evento e;
  for (uint8_t i = 0; i < MAX_LOG; i++) {
    EEPROM.get(INICIO_LOG + i * sizeof(Evento), e);
    if (e.tipo == 0xFF) continue;
    Serial.print(i); Serial.print(F(" | ")); Serial.print(e.cartao); Serial.print(F(" | "));
    Serial.print(e.tipo == EV_ABRIU ? F("abriu") : e.tipo == EV_NEGADO ? F("negado") : F("cadastro"));
    Serial.print(F(" | ")); Serial.println(e.segundos);
  }
}

void listar() {
  Serial.println(F("--- cartoes ---"));
  Cartao c;
  for (uint8_t i = 0; i < MAX_CARTOES; i++) {
    lerRegistro(i, c);
    if (registroLivre(c)) continue;
    Serial.print(i); Serial.print(F(": "));
    for (uint8_t k = 0; k < 4; k++) { if (c.uid[k] < 16) Serial.print('0'); Serial.print(c.uid[k], HEX); }
    Serial.print(F(" perm=")); Serial.print(c.perm, BIN);
    Serial.print(F(" acessos=")); Serial.print(c.acessos);
    Serial.println(c.soma == calcularSoma(c) ? F(" ok") : F(" CORROMPIDO"));
  }
}

void sinal(bool bom) {
  digitalWrite(bom ? PIN_VERDE : PIN_VERMELHO, HIGH);
  tone(PIN_BUZZER, bom ? 1500 : 300, bom ? 120 : 400);
  delay(300);
  digitalWrite(PIN_VERDE, LOW); digitalWrite(PIN_VERMELHO, LOW);
}

void abrirTrava() {
  trava.attach(PIN_SERVO);
  trava.write(90);
  digitalWrite(PIN_VERDE, HIGH);
  delay(3000);
  trava.write(0);
  delay(400);
  trava.detach();
  digitalWrite(PIN_VERDE, LOW);
}

bool cadastrarNovo(const uint8_t *uid) {
  Cartao c;
  for (uint8_t i = 0; i < MAX_CARTOES; i++) {
    lerRegistro(i, c);
    if (registroLivre(c)) {
      memcpy(c.uid, uid, 4); c.perm = PERM_ABRIR; c.acessos = 0;
      gravarRegistro(i, c);
      registrarEvento(i, EV_CADASTRO);
      Serial.print(F("Cartao cadastrado na posicao ")); Serial.println(i);
      return true;
    }
  }
  return false;
}

void tratarCartao(const uint8_t *uid) {
  int idx = buscar(uid);
  Cartao c;

  if (modo == 'a') {                                  // cadastro pelo Serial
    modo = 'u';
    if (idx >= 0) { Serial.println(F("Ja cadastrado.")); sinal(false); }
    else if (cadastrarNovo(uid)) sinal(true); else { Serial.println(F("Tabela cheia")); sinal(false); }
    return;
  }

  if (modo == 'j' && millis() < ate) {                // janela aberta pelo administrador
    modo = 'u';
    if (idx >= 0) { Serial.println(F("Ja cadastrado.")); sinal(false); }
    else if (cadastrarNovo(uid)) sinal(true); else { Serial.println(F("Tabela cheia")); sinal(false); }
    return;
  }
  modo = 'u';

  if (idx < 0) { Serial.println(F("ACESSO NEGADO (cartao desconhecido)")); registrarEvento(255, EV_NEGADO); sinal(false); return; }

  lerRegistro(idx, c);
  if (c.perm & PERM_ADMIN) {                          // administrador: abre a janela de cadastro
    modo = 'j'; ate = millis() + JANELA_ADMIN_MS;
    Serial.println(F("ADMIN reconhecido: passe o cartao NOVO em ate 5 s"));
    tone(PIN_BUZZER, 2000, 80); delay(150); tone(PIN_BUZZER, 2000, 80);
    return;
  }
  if (c.perm & PERM_ABRIR) {
    c.acessos++;
    gravarRegistro(idx, c);
    registrarEvento(idx, EV_ABRIU);
    Serial.print(F("ACESSO PERMITIDO | acessos: ")); Serial.println(c.acessos);
    abrirTrava();
  } else {
    Serial.println(F("ACESSO NEGADO (sem permissao)"));
    registrarEvento(idx, EV_NEGADO);
    sinal(false);
  }
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_VERDE, OUTPUT); pinMode(PIN_VERMELHO, OUTPUT); pinMode(PIN_BUZZER, OUTPUT);
  trava.attach(PIN_SERVO); trava.write(0); delay(400); trava.detach();
  SPI.begin();
  leitor.PCD_Init();
  if (EEPROM.read(0) != MAGICO) { Serial.println(F("EEPROM sem formato: formatando...")); formatar(); }
  descobrirPonteiroLog();
  Serial.println(F("=== COFRE RFID - Fase 3 ==="));
  Serial.println(F("Comandos: l cartoes | g log | a cadastrar | m 1o cartao vira admin | f formatar"));
  Serial.print(F("sizeof(Evento) = ")); Serial.print(sizeof(Evento)); Serial.println(F(" bytes"));
  listar();
}

void loop() {
  while (Serial.available()) {
    char ch = Serial.read();
    if (ch == 'l') listar();
    if (ch == 'g') mostrarLog();
    if (ch == 'a') { modo = 'a'; Serial.println(F(">> aproxime o cartao a CADASTRAR")); }
    if (ch == 'f') formatar();
    if (ch == 'm') {
      Cartao c; lerRegistro(0, c);
      if (!registroLivre(c)) { c.perm = PERM_ABRIR | PERM_ADMIN; gravarRegistro(0, c); Serial.println(F(">> cartao 0 agora e administrador")); }
    }
  }
  if (!leitor.PICC_IsNewCardPresent() || !leitor.PICC_ReadCardSerial()) return;
  tratarCartao(leitor.uid.uidByte);
  leitor.PICC_HaltA();
}
