Missão 1 · Montagem

🔌 Monte o circuito

Separe os materiais e faça as ligações antes de carregar o primeiro código.

🧰 Materiais

  • 1 Arduino Uno R3 + cabo USB
  • 1 protoboard 830 pontos
  • Até 9 transistores NPN BC337 ou PN2222A
  • Resistores de 10 kΩ (1 por entrada de transistor, até 9), de 1 kΩ (1 por saída de porta, até 5) e de 300 Ω (4, para os LEDs)
  • 4 LEDs
  • Jumpers
  • Opcional: multímetro

⚠️ Antes de começar

  • Pinagem do transistor: confira no desenho do seu modelo (BC337: E-B-C vira C-B-E conforme o lado; PN2222A tem outra ordem). Ligar errado não queima na hora, mas o circuito não funciona.
  • Fases 2 e 3: NOR = 2 transistores em PARALELO (coletores juntos); NAND = 2 transistores em SÉRIE (emissor de um no coletor do outro). Veja as descrições no código.
  • Meio-somador (fase 3): N1 = NAND(A,B); N2 = NAND(A,N1); N3 = NAND(B,N1); SOMA = NAND(N2,N3); VAI-UM = NOT(N1). São 4 NAND + 1 inversor = 9 transistores.
  • Os tempos de propagação medidos dependem da montagem e do transistor; o objetivo é comparar subida × descida.

Missão 2 · Fase 1

🔁 O inversor (NOT) com transistor

Montar um inversor com um transistor NPN. O Arduino aplica A, mede Y em milivolts e classifica o nível lógico (BAIXO, ALTO, INDEFINIDO).

💻 Código da fase 1 portas1_inversor.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  LABORATORIO DE PORTAS LOGICAS - FASE 1: o inversor (NOT) feito com transistor

  Um transistor NPN (BC337 ou PN2222A) funciona como uma CHAVE comandada por tensao.
  Montagem do inversor (RTL):
      entrada A ---[ 10 kohms ]--- base do transistor
      emissor do transistor ------ GND
      coletor do transistor ------ saida Y   e   Y ---[ 1 kohm ]--- 5V
  Entrada ALTA  -> transistor liga  -> Y ~ 0 V (BAIXO)
  Entrada BAIXA -> transistor desliga -> Y = 5 V (ALTO)   =>  Y = NOT A

  O Arduino faz de "gerador de sinais" (aplica A) e de "analisador logico" (le Y).

  Ligacoes ao Arduino:
    pino 2  -> entrada A do inversor (pelo resistor de 10 kohms)
    A0      -> saida Y do inversor      (le a tensao em mV)
    GND do Arduino ligado ao GND da placa de ensaio
    LEDs de monitoramento (300 ohms): pino 8 = A, pino 10 = Y lido pelo Arduino
*/
const uint8_t PIN_A = 2;
const uint8_t PIN_LED_A = 8;
const uint8_t PIN_LED_Y = 10;
const uint8_t PIN_Y = A0;

// Faixas de tensao tipicas de uma entrada TTL/CMOS de 5 V
const int VIL_MV = 1500;    // abaixo disso e BAIXO
const int VIH_MV = 3000;    // acima disso e ALTO

int lerMilivolts(uint8_t pino) {
  long soma = 0;
  for (uint8_t i = 0; i < 8; i++) soma += analogRead(pino);
  return (int)(soma / 8 * 5000L / 1023L);
}

const __FlashStringHelper* classificar(int mv) {
  if (mv < VIL_MV) return F("BAIXO (0)");
  if (mv > VIH_MV) return F("ALTO (1)");
  return F("INDEFINIDO");
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_A, OUTPUT);
  pinMode(PIN_LED_A, OUTPUT);
  pinMode(PIN_LED_Y, OUTPUT);
  Serial.println(F("=== PORTAS LOGICAS - Fase 1: inversor (NOT) ==="));
  Serial.println(F("A | Y medido (mV) | nivel logico   | esperado | resultado"));
}

void loop() {
  uint8_t acertos = 0;
  for (uint8_t a = 0; a <= 1; a++) {
    digitalWrite(PIN_A, a);
    digitalWrite(PIN_LED_A, a);
    delay(50);
    int mv = lerMilivolts(PIN_Y);
    bool y = mv > 2500;
    digitalWrite(PIN_LED_Y, y);

    bool esperado = !a;
    bool ok = (y == esperado);
    if (ok) acertos++;

    Serial.print(a); Serial.print(F(" | ")); Serial.print(mv); Serial.print(F(" mV        | "));
    Serial.print(classificar(mv)); Serial.print(F(" | ")); Serial.print(esperado);
    Serial.print(F("        | ")); Serial.println(ok ? F("OK") : F("ERRO"));
    delay(800);
  }
  Serial.print(F("Linhas corretas: ")); Serial.print(acertos); Serial.println(F("/2"));
  Serial.println(F("-----"));
  delay(1500);
}
Missão 3 · Fase 2

🔀 NOR, NAND e o tempo de propagação

Montar NOR (paralelo) e NAND (série), conferir a tabela-verdade com o Arduino e medir o atraso entre a mudança da entrada e a saída com o Timer1 (62,5 ns por tick).

💻 Código da fase 2 portas2_nor_nand.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  LABORATORIO DE PORTAS LOGICAS - FASE 2: NOR e NAND + tempo de propagacao

  NOR (2 transistores em PARALELO)  : Y = NOT (A OR B)
     cada entrada -> 10 kohms -> base de um transistor; os dois emissores no GND;
     os dois coletores juntos formam Y; Y ---[ 1 kohm ]--- 5V.
  NAND (2 transistores em SERIE)     : Y = NOT (A AND B)
     entrada A -> 10 kohms -> base do T1;  entrada B -> 10 kohms -> base do T2;
     coletor do T1 = Y;  emissor do T1 -> coletor do T2;  emissor do T2 -> GND;
     Y ---[ 1 kohm ]--- 5V.
  NOR e NAND sao portas UNIVERSAIS: qualquer circuito digital pode ser feito so com elas.

  Comandos no Monitor Serial:  o = testar NOR | a = testar NAND | i = testar NOT (so A)
  O programa confere a tabela-verdade e mede o TEMPO DE PROPAGACAO com o Timer1
  (cada tick do contador vale 62,5 ns a 16 MHz).

  Ligacoes ao Arduino:
    pino 2 -> entrada A | pino 3 -> entrada B
    saida Y -> pino 4 (leitura digital) E pino A0 (leitura analogica) ao mesmo tempo
    LEDs (300 ohms): pino 8 = A, 9 = B, 10 = Y lido
*/
const uint8_t PIN_A = 2, PIN_B = 3, PIN_Y = 4;
const uint8_t PIN_LED_A = 8, PIN_LED_B = 9, PIN_LED_Y = 10;
char porta = 'o';

bool esperado(char tipo, bool a, bool b) {
  if (tipo == 'o') return !(a || b);
  if (tipo == 'a') return !(a && b);
  return !a;
}

const __FlashStringHelper* nomePorta(char tipo) {
  if (tipo == 'o') return F("NOR");
  if (tipo == 'a') return F("NAND");
  return F("NOT");
}

void aplicar(bool a, bool b) {
  digitalWrite(PIN_A, a); digitalWrite(PIN_B, b);
  digitalWrite(PIN_LED_A, a); digitalWrite(PIN_LED_B, b);
}

// Mede quanto tempo (em ns) a saida demora para chegar ao valor esperado
// depois que as entradas mudam. Usa o Timer1 e leitura direta da porta D.
long medirAtrasoNs(bool aIni, bool bIni, bool aFim, bool bFim, bool yFinal) {
  aplicar(aIni, bIni);
  delay(5);
  noInterrupts();
  TCCR1A = 0; TCCR1B = 0; TCNT1 = 0;
  TCCR1B = (1 << CS10);                         // contador a 16 MHz, sem prescaler
  if (aFim) PORTD |= (1 << PD2); else PORTD &= ~(1 << PD2);
  if (bFim) PORTD |= (1 << PD3); else PORTD &= ~(1 << PD3);
  uint16_t t0 = TCNT1;
  uint16_t t1 = t0;
  uint8_t alvo = yFinal ? (1 << PD4) : 0;
  while ((PIND & (1 << PD4)) != alvo) {
    t1 = TCNT1;
    if ((uint16_t)(t1 - t0) > 60000) break;     // desiste (timeout ~3,7 ms)
  }
  t1 = TCNT1;
  interrupts();
  TCCR1B = 0;
  uint16_t ticks = t1 - t0;
  return (long)ticks * 625L / 10L;              // ticks * 62,5 ns
}

void testar() {
  Serial.print(F("--- Porta: ")); Serial.print(nomePorta(porta)); Serial.println(F(" ---"));
  Serial.println(F("A B | Y (mV) | Y lido | esperado | resultado"));
  uint8_t acertos = 0, total = 0;
  for (uint8_t a = 0; a <= 1; a++) {
    for (uint8_t b = 0; b <= 1; b++) {
      if (porta == 'i' && b == 1) continue;     // NOT so usa A
      aplicar(a, b);
      delay(40);
      int mv = (int)((long)analogRead(A0) * 5000L / 1023L);
      bool y = digitalRead(PIN_Y);
      digitalWrite(PIN_LED_Y, y);
      bool e = esperado(porta, a, b);
      bool ok = (y == e);
      total++; if (ok) acertos++;
      Serial.print(a); Serial.print(' '); Serial.print(b); Serial.print(F(" | "));
      Serial.print(mv); Serial.print(F("    | ")); Serial.print(y);
      Serial.print(F("      | ")); Serial.print(e);
      Serial.print(F("        | ")); Serial.println(ok ? F("OK") : F("ERRO"));
      delay(300);
    }
  }
  Serial.print(F("Linhas corretas: ")); Serial.print(acertos); Serial.print('/'); Serial.println(total);

  // Atraso de propagacao: de (0,0) para (1,1) a saida deve cair (NOR/NAND/NOT) ...
  long descida = medirAtrasoNs(0, 0, 1, 1, esperado(porta, 1, 1));
  // ... e de (1,1) para (0,0) a saida deve subir.
  long subida = medirAtrasoNs(1, 1, 0, 0, esperado(porta, 0, 0));
  Serial.print(F("atraso (0,0)->(1,1): ")); Serial.print(descida); Serial.println(F(" ns"));
  Serial.print(F("atraso (1,1)->(0,0): ")); Serial.print(subida); Serial.println(F(" ns"));
  Serial.println(F("(a subida costuma ser mais lenta: o resistor de 1 kohm carrega a capacitancia da saida)"));
  aplicar(0, 0);
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_A, OUTPUT); pinMode(PIN_B, OUTPUT); pinMode(PIN_Y, INPUT);
  pinMode(PIN_LED_A, OUTPUT); pinMode(PIN_LED_B, OUTPUT); pinMode(PIN_LED_Y, OUTPUT);
  Serial.println(F("=== PORTAS LOGICAS - Fase 2 ==="));
  Serial.println(F("Comandos: o = NOR | a = NAND | i = NOT"));
  testar();
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'o' || c == 'a' || c == 'i') { porta = c; testar(); }
  }
}
Missão 4 · Fase 3

➕ O meio-somador de 9 transistores

Montar um meio-somador (SOMA = A XOR B, VAI-UM = A AND B) só com portas NAND e um inversor, e conferir a tabela-verdade com o Arduino.

💻 Código da fase 3 portas3_meio_somador.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  LABORATORIO DE PORTAS LOGICAS - FASE 3: o MEIO-SOMADOR feito de portas NAND

  Meio-somador (soma de 2 bits):   SOMA = A XOR B     VAI-UM = A AND B
  Com 4 portas NAND montamos o XOR:
      N1 = NAND(A, B)
      N2 = NAND(A, N1)
      N3 = NAND(B, N1)
      SOMA = NAND(N2, N3)         (isto e A XOR B)
  E o VAI-UM e apenas o inverso de N1:
      VAI-UM = NOT(N1)            (um inversor, a Fase 1)

  Total: 4 portas NAND + 1 inversor = 9 transistores NPN.

  Ligacoes ao Arduino:
    pino 2 -> entrada A | pino 3 -> entrada B
    saida SOMA   -> pino 4
    saida VAI-UM -> pino 5
    LEDs (300 ohms): pino 8 = A, 9 = B, 10 = SOMA, 11 = VAI-UM
*/
const uint8_t PIN_A = 2, PIN_B = 3, PIN_SOMA = 4, PIN_CARRY = 5;
const uint8_t PIN_LED_A = 8, PIN_LED_B = 9, PIN_LED_SOMA = 10, PIN_LED_CARRY = 11;

void setup() {
  Serial.begin(9600);
  pinMode(PIN_A, OUTPUT); pinMode(PIN_B, OUTPUT);
  pinMode(PIN_SOMA, INPUT); pinMode(PIN_CARRY, INPUT);
  pinMode(PIN_LED_A, OUTPUT); pinMode(PIN_LED_B, OUTPUT);
  pinMode(PIN_LED_SOMA, OUTPUT); pinMode(PIN_LED_CARRY, OUTPUT);
  Serial.println(F("=== PORTAS LOGICAS - Fase 3: meio-somador ==="));
}

void loop() {
  uint8_t acertos = 0;
  Serial.println(F("A B | SOMA VAI-UM | A+B (binario) | esperado | resultado"));
  for (uint8_t a = 0; a <= 1; a++) {
    for (uint8_t b = 0; b <= 1; b++) {
      digitalWrite(PIN_A, a); digitalWrite(PIN_B, b);
      digitalWrite(PIN_LED_A, a); digitalWrite(PIN_LED_B, b);
      delay(60);
      bool soma = digitalRead(PIN_SOMA);
      bool carry = digitalRead(PIN_CARRY);
      digitalWrite(PIN_LED_SOMA, soma);
      digitalWrite(PIN_LED_CARRY, carry);

      uint8_t esperadoSoma = a ^ b;
      uint8_t esperadoCarry = a & b;
      bool ok = (soma == esperadoSoma) && (carry == esperadoCarry);
      if (ok) acertos++;

      Serial.print(a); Serial.print(' '); Serial.print(b); Serial.print(F(" |  "));
      Serial.print(soma); Serial.print(F("      ")); Serial.print(carry);
      Serial.print(F("    |      ")); Serial.print(carry); Serial.print(soma);
      Serial.print(F("        |   ")); Serial.print(esperadoCarry); Serial.print(esperadoSoma);
      Serial.print(F("     | ")); Serial.println(ok ? F("OK") : F("ERRO"));
      delay(900);
    }
  }
  Serial.print(F("Linhas corretas: ")); Serial.print(acertos); Serial.println(F("/4"));
  Serial.println(F("-----"));
  delay(1500);
}
