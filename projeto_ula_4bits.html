Missão 1 · Montagem

🔌 Monte o circuito

Separe os materiais e faça as ligações antes de carregar o primeiro código.

🧰 Materiais

  • 1 Arduino Uno R3 + cabo USB
  • 1 protoboard 830 pontos
  • 1 teclado matricial de membrana 4×4
  • 4 LEDs: verde, amarelo, vermelho e azul
  • 1 LED RGB (catodo comum; veja o aviso)
  • 7 resistores de 300 Ω (4 para os LEDs e 3 para o RGB)
  • Jumpers macho-macho e macho-fêmea

⚠️ Antes de começar

  • O LED RGB pode ser de catodo comum (pino longo no GND) ou anodo comum (pino longo no 5V). Se as cores ficarem invertidas, troque COMUM_ANODO nas fases 2 e 3.
  • Se alguma tecla “falhar”, confira se as linhas e colunas não estão trocadas: a ordem dos 8 fios do teclado é L1, L2, L3, L4, C1, C2, C3, C4 (veja o desenho do seu modelo).
  • No Tinkercad existe o teclado 4×4 e o LED RGB; as ligações são as mesmas.

Missão 2 · Fase 1

⌨️ Teclado, hexadecimal e complemento de dois

Ler o teclado por varredura e mostrar cada tecla como número de 4 bits: nos LEDs, em hexadecimal, binário, decimal sem sinal e decimal com sinal (complemento de dois).

💻 Código da fase 1 ula1_teclado_hex.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  MINI-ULA DE 4 BITS - FASE 1: teclado matricial e numeros hexadecimais

  Um teclado 4x4 tem 16 teclas = 16 valores = 1 digito HEXADECIMAL (4 bits).
    Teclas 0-9 -> 0 a 9 | A,B,C,D -> 10 a 13 | * -> 14 (E) | # -> 15 (F)

  Como o teclado funciona (varredura/multiplexacao):
    4 linhas (saidas) + 4 colunas (entradas com pull-up) = 8 fios para 16 teclas.
    O Arduino poe UMA linha em LOW de cada vez e olha qual coluna caiu para LOW.

  Ligacoes:
    Linhas do teclado  -> pinos 2, 3, 4, 5
    Colunas do teclado -> pinos 6, 7, 8, 9
    LEDs (cada um com resistor de 300 ohms para o GND):
      bit 0 = LED verde no pino 13   bit 1 = LED amarelo no pino 12
      bit 2 = LED vermelho no pino 11  bit 3 = LED azul no pino 10
*/
const uint8_t LINHAS[4]  = {2, 3, 4, 5};
const uint8_t COLUNAS[4] = {6, 7, 8, 9};
const uint8_t LEDS[4]    = {13, 12, 11, 10};   // bit 0 ... bit 3
const char TECLAS[4][4]  = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};

void iniciarTeclado() {
  for (uint8_t r = 0; r < 4; r++) { pinMode(LINHAS[r], OUTPUT); digitalWrite(LINHAS[r], HIGH); }
  for (uint8_t c = 0; c < 4; c++) pinMode(COLUNAS[c], INPUT_PULLUP);
}

// Varredura: devolve a tecla pressionada ou 0
char lerTecla() {
  for (uint8_t r = 0; r < 4; r++) {
    digitalWrite(LINHAS[r], LOW);
    for (uint8_t c = 0; c < 4; c++) {
      if (digitalRead(COLUNAS[c]) == LOW) {
        digitalWrite(LINHAS[r], HIGH);
        return TECLAS[r][c];
      }
    }
    digitalWrite(LINHAS[r], HIGH);
  }
  return 0;
}

// Espera apertar, espera soltar (anti-repique) e devolve a tecla
char esperarTecla() {
  char t;
  while ((t = lerTecla()) == 0) { }
  delay(30);
  while (lerTecla() != 0) { }
  delay(30);
  return t;
}

int8_t valorHex(char t) {
  if (t >= '0' && t <= '9') return t - '0';
  if (t >= 'A' && t <= 'D') return 10 + (t - 'A');
  if (t == '*') return 14;
  if (t == '#') return 15;
  return -1;
}

void mostrar4bits(uint8_t v) {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (v >> b) & 1);
}

void imprimirBinario(uint8_t v, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((v >> i) & 1);
}

void setup() {
  Serial.begin(9600);
  iniciarTeclado();
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  Serial.println(F("=== MINI-ULA - Fase 1: teclado hexadecimal ==="));
  Serial.println(F("Aperte uma tecla. * = E e # = F."));
}

void loop() {
  char tecla = esperarTecla();
  uint8_t v = valorHex(tecla);
  mostrar4bits(v);

  int8_t comSinal = (v & 0b1000) ? (int8_t)v - 16 : (int8_t)v;   // complemento de dois em 4 bits

  Serial.print(F("tecla '")); Serial.print(tecla);
  Serial.print(F("' | HEX = ")); Serial.print(v, HEX);
  Serial.print(F(" | BIN = ")); imprimirBinario(v, 4);
  Serial.print(F(" | DEC sem sinal = ")); Serial.print(v);
  Serial.print(F(" | DEC com sinal (compl. de 2) = ")); Serial.println(comSinal);
}
Missão 3 · Fase 2

🧮 A ULA: operações, opcodes e flags

Montar uma “instrução” de 3 teclas (A, B, opcode) e executar a operação: ADD, SUB, AND, OR, XOR, NOT, deslocamentos. Os LEDs mostram o resultado e o RGB mostra as flags C, Z e N.

💻 Código da fase 2 ula2_operacoes.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  MINI-ULA DE 4 BITS - FASE 2: a Unidade Logica e Aritmetica (ULA)

  Uma "instrucao" tem 3 teclas (como um mini codigo de maquina):
      1a tecla = operando A    2a tecla = operando B    3a tecla = OPCODE
  Opcodes (hexadecimal):
      0 = A + B    1 = A - B    2 = A AND B   3 = A OR B
      4 = A XOR B  5 = NOT A    6 = A << 1    7 = A >> 1
  O resultado aparece nos 4 LEDs. O LED RGB mostra os SINAIS (flags):
      VERMELHO = C (vai-um / carry)   VERDE = Z (resultado zero)   AZUL = N (negativo)
  A flag V (overflow, estouro com sinal) aparece no Monitor Serial.

  Ligacoes: as mesmas da Fase 1, mais o LED RGB (cada cor com 300 ohms):
      cor vermelha -> A0   cor verde -> A1   cor azul -> A2
      (catodo comum: pino longo no GND. Se o seu for anodo comum, mude COMUM_ANODO)
*/
const uint8_t LINHAS[4]  = {2, 3, 4, 5};
const uint8_t COLUNAS[4] = {6, 7, 8, 9};
const uint8_t LEDS[4]    = {13, 12, 11, 10};
const uint8_t PIN_RGB[3] = {A0, A1, A2};
const bool COMUM_ANODO   = false;
const char TECLAS[4][4]  = {
  {'1', '2', '3', 'A'}, {'4', '5', '6', 'B'}, {'7', '8', '9', 'C'}, {'*', '0', '#', 'D'}
};

uint8_t resultado, flagC, flagZ, flagN, flagV;

void iniciarTeclado() {
  for (uint8_t r = 0; r < 4; r++) { pinMode(LINHAS[r], OUTPUT); digitalWrite(LINHAS[r], HIGH); }
  for (uint8_t c = 0; c < 4; c++) pinMode(COLUNAS[c], INPUT_PULLUP);
}
char lerTecla() {
  for (uint8_t r = 0; r < 4; r++) {
    digitalWrite(LINHAS[r], LOW);
    for (uint8_t c = 0; c < 4; c++) {
      if (digitalRead(COLUNAS[c]) == LOW) { digitalWrite(LINHAS[r], HIGH); return TECLAS[r][c]; }
    }
    digitalWrite(LINHAS[r], HIGH);
  }
  return 0;
}
char esperarTecla() {
  char t;
  while ((t = lerTecla()) == 0) { }
  delay(30);
  while (lerTecla() != 0) { }
  delay(30);
  return t;
}
uint8_t valorHex(char t) {
  if (t >= '0' && t <= '9') return t - '0';
  if (t >= 'A' && t <= 'D') return 10 + (t - 'A');
  if (t == '*') return 14;
  return 15;
}
void imprimirBinario(uint8_t v, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((v >> i) & 1);
}
void mostrarLeds(uint8_t v) {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (v >> b) & 1);
}
void mostrarFlags() {
  uint8_t f[3] = {flagC, flagZ, flagN};
  for (uint8_t i = 0; i < 3; i++) digitalWrite(PIN_RGB[i], COMUM_ANODO ? !f[i] : f[i]);
}

// ---------- A ULA ----------
void ula(uint8_t a, uint8_t b, uint8_t op) {
  uint8_t r = 0, c = 0, v = 0;
  switch (op) {
    case 0: {                                   // soma
      uint8_t s = a + b;
      r = s & 0xF; c = s >> 4;
      v = (((a ^ r) & (b ^ r)) & 0b1000) ? 1 : 0;
    } break;
    case 1: {                                   // subtracao = A + (NOT B) + 1  (complemento de dois)
      uint8_t s = a + ((~b) & 0xF) + 1;
      r = s & 0xF; c = s >> 4;
      v = (((a ^ b) & (a ^ r)) & 0b1000) ? 1 : 0;
    } break;
    case 2: r = a & b; break;
    case 3: r = a | b; break;
    case 4: r = a ^ b; break;
    case 5: r = (~a) & 0xF; break;
    case 6: c = (a >> 3) & 1; r = (a << 1) & 0xF; break;
    case 7: c = a & 1;        r = a >> 1;         break;
    default: r = 0; break;
  }
  resultado = r; flagC = c; flagV = v;
  flagZ = (r == 0);
  flagN = (r >> 3) & 1;
}

const char* nomeOp(uint8_t op) {
  const char* nomes[8] = {"ADD", "SUB", "AND", "OR", "XOR", "NOT A", "SHL A", "SHR A"};
  return op < 8 ? nomes[op] : "???";
}

void setup() {
  Serial.begin(9600);
  iniciarTeclado();
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  for (uint8_t i = 0; i < 3; i++) pinMode(PIN_RGB[i], OUTPUT);
  Serial.println(F("=== MINI-ULA - Fase 2 ==="));
  Serial.println(F("Digite: A, B, opcode (0 ADD 1 SUB 2 AND 3 OR 4 XOR 5 NOT 6 SHL 7 SHR)"));
}

void loop() {
  Serial.println(F("--- nova instrucao ---"));
  uint8_t a = valorHex(esperarTecla());  Serial.print(F("A = "));  Serial.println(a, HEX);
  uint8_t b = valorHex(esperarTecla());  Serial.print(F("B = "));  Serial.println(b, HEX);
  uint8_t op = valorHex(esperarTecla()); Serial.print(F("opcode = ")); Serial.println(op, HEX);

  ula(a, b, op);
  mostrarLeds(resultado);
  mostrarFlags();

  imprimirBinario(a, 4); Serial.print(' '); Serial.print(nomeOp(op)); Serial.print(' ');
  imprimirBinario(b, 4); Serial.print(F(" = ")); imprimirBinario(resultado, 4);
  Serial.print(F("  (HEX ")); Serial.print(resultado, HEX); Serial.println(')');
  Serial.print(F("flags: C=")); Serial.print(flagC);
  Serial.print(F(" Z="));       Serial.print(flagZ);
  Serial.print(F(" N="));       Serial.print(flagN);
  Serial.print(F(" V="));       Serial.println(flagV);
}
Missão 4 · Fase 3

🔧 ULA só com portas lógicas e acumulador

Trocar a soma “pronta” por um somador/subtrator de 4 bits montado com portas XOR, AND e OR (ripple carry), validar tudo com um autoteste de 512 operações e usar um registrador acumulador (ACC).

💻 Código da fase 3 ula3_portas_acumulador.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  MINI-ULA DE 4 BITS - FASE 3: ULA feita so com portas logicas + acumulador

  Novidades:
    - A soma e a subtracao usam um SOMADOR/SUBTRATOR montado bit a bit com
      portas logicas (XOR, AND, OR), igual ao hardware de verdade.
      Subtracao: troca B por NOT B (XOR com 1) e entra com carry-in = 1.
    - Comando 't' no Monitor Serial: AUTOTESTE com as 256 combinacoes de A e B.
    - Comando 'a': liga/desliga o MODO ACUMULADOR. Nesse modo a ULA guarda o
      resultado no registrador ACC e a proxima instrucao usa ACC como operando A
      (so digite B e o opcode). Comando 'z': zera o ACC.

  Ligacoes: as mesmas da Fase 2.
*/
const uint8_t LINHAS[4]  = {2, 3, 4, 5};
const uint8_t COLUNAS[4] = {6, 7, 8, 9};
const uint8_t LEDS[4]    = {13, 12, 11, 10};
const uint8_t PIN_RGB[3] = {A0, A1, A2};
const bool COMUM_ANODO   = false;
const char TECLAS[4][4]  = {
  {'1', '2', '3', 'A'}, {'4', '5', '6', 'B'}, {'7', '8', '9', 'C'}, {'*', '0', '#', 'D'}
};

uint8_t resultado, flagC, flagZ, flagN, flagV;
uint8_t acc = 0;
bool modoAcumulador = false;

void iniciarTeclado() {
  for (uint8_t r = 0; r < 4; r++) { pinMode(LINHAS[r], OUTPUT); digitalWrite(LINHAS[r], HIGH); }
  for (uint8_t c = 0; c < 4; c++) pinMode(COLUNAS[c], INPUT_PULLUP);
}
char lerTecla() {
  for (uint8_t r = 0; r < 4; r++) {
    digitalWrite(LINHAS[r], LOW);
    for (uint8_t c = 0; c < 4; c++) {
      if (digitalRead(COLUNAS[c]) == LOW) { digitalWrite(LINHAS[r], HIGH); return TECLAS[r][c]; }
    }
    digitalWrite(LINHAS[r], HIGH);
  }
  return 0;
}
// Igual a Fase 2, mas tambem atende comandos do Serial enquanto espera
char esperarTecla() {
  char t;
  while ((t = lerTecla()) == 0) {
    if (Serial.available()) return 0;
  }
  delay(30);
  while (lerTecla() != 0) { }
  delay(30);
  return t;
}
uint8_t valorHex(char t) {
  if (t >= '0' && t <= '9') return t - '0';
  if (t >= 'A' && t <= 'D') return 10 + (t - 'A');
  if (t == '*') return 14;
  return 15;
}
void imprimirBinario(uint8_t v, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((v >> i) & 1);
}
void mostrarLeds(uint8_t v) {
  for (uint8_t b = 0; b < 4; b++) digitalWrite(LEDS[b], (v >> b) & 1);
}

  for (uint8_t i = 0; i < 3; i++) digitalWrite(PIN_RGB[i], COMUM_ANODO ? !f[i] : f[i]);

// ---------- Somador completo de 1 bit ----------
void somaCompleta(uint8_t a, uint8_t b, uint8_t cin, uint8_t &s, uint8_t &cout) {
  s    = a ^ b ^ cin;
  cout = (a & b) | (cin & (a ^ b));
}

// ---------- Somador/Subtrator de 4 bits (ripple carry) ----------
// sub = 0 -> A + B      sub = 1 -> A + (B XOR 1111) + 1  = A - B
uint8_t somadorSubtrator(uint8_t a, uint8_t b, uint8_t sub, uint8_t &carryFinal, uint8_t &overflow) {
  uint8_t carry = sub, r = 0, carryAntesDoUltimo = 0;
  for (uint8_t i = 0; i < 4; i++) {
    uint8_t bi = ((b >> i) & 1) ^ sub;          // XOR com "sub" inverte B na subtracao
    uint8_t s;
    if (i == 3) carryAntesDoUltimo = carry;
    somaCompleta((a >> i) & 1, bi, carry, s, carry);
    r |= s << i;
  }
  carryFinal = carry;
  overflow = carry ^ carryAntesDoUltimo;        // overflow = carry que entra XOR carry que sai do bit de sinal
  return r;
}

void ula(uint8_t a, uint8_t b, uint8_t op) {
  uint8_t r = 0, c = 0, v = 0;
  switch (op) {
    case 0: r = somadorSubtrator(a, b, 0, c, v); break;
    case 1: r = somadorSubtrator(a, b, 1, c, v); break;
    case 2: r = a & b; break;
    case 3: r = a | b; break;
    case 4: r = a ^ b; break;
    case 5: r = (~a) & 0xF; break;
    case 6: c = (a >> 3) & 1; r = (a << 1) & 0xF; break;
    case 7: c = a & 1;        r = a >> 1;         break;
    default: r = 0; break;
  }
  resultado = r; flagC = c; flagV = v;
  flagZ = (r == 0);
  flagN = (r >> 3) & 1;
}

// Referencia "de verdade" (aritmetica do C) para conferir o hardware simulado
uint8_t referenciaSoma(uint8_t a, uint8_t b) { return (a + b) & 0xF; }
uint8_t referenciaSub(uint8_t a, uint8_t b)  { return (a - b) & 0xF; }

void autoteste() {
  uint16_t erros = 0;
  for (uint8_t a = 0; a < 16; a++) {
    for (uint8_t b = 0; b < 16; b++) {
      ula(a, b, 0); if (resultado != referenciaSoma(a, b)) erros++;
      ula(a, b, 1); if (resultado != referenciaSub(a, b))  erros++;
    }
  }
  Serial.print(F("AUTOTESTE: 512 operacoes, erros = "));
  Serial.println(erros);
  if (erros == 0) Serial.println(F("A ULA de portas logicas esta CORRETA!"));
}

void tratarComandosSerial() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 't') autoteste();
    if (c == 'a') { modoAcumulador = !modoAcumulador; Serial.print(F(">> modo acumulador: ")); Serial.println(modoAcumulador ? F("LIGADO") : F("desligado")); }
    if (c == 'z') { acc = 0; Serial.println(F(">> ACC = 0")); }
  }
}

void setup() {
  Serial.begin(9600);
  iniciarTeclado();
  for (uint8_t b = 0; b < 4; b++) pinMode(LEDS[b], OUTPUT);
  for (uint8_t i = 0; i < 3; i++) pinMode(PIN_RGB[i], OUTPUT);
  Serial.println(F("=== MINI-ULA - Fase 3 ==="));
  Serial.println(F("Serial: t = autoteste | a = modo acumulador | z = zera ACC"));
}

void loop() {
  uint8_t a, b, op;
  char t;

  if (modoAcumulador) {
    a = acc;
    Serial.print(F("ACC = ")); Serial.println(a, HEX);
  } else {
    t = esperarTecla(); if (t == 0) { tratarComandosSerial(); return; }
    a = valorHex(t); Serial.print(F("A = ")); Serial.println(a, HEX);
  }
  t = esperarTecla(); if (t == 0) { tratarComandosSerial(); return; }
  b = valorHex(t); Serial.print(F("B = ")); Serial.println(b, HEX);
  t = esperarTecla(); if (t == 0) { tratarComandosSerial(); return; }
  op = valorHex(t); Serial.print(F("opcode = ")); Serial.println(op, HEX);

  ula(a, b, op);
  mostrarLeds(resultado);
  mostrarFlags();
  if (modoAcumulador) acc = resultado;       // o resultado volta para o registrador ACC

  imprimirBinario(a, 4); Serial.print(F(" op")); Serial.print(op, HEX); Serial.print(' ');
  imprimirBinario(b, 4); Serial.print(F(" = ")); imprimirBinario(resultado, 4);
  Serial.print(F(" | C=")); Serial.print(flagC);
  Serial.print(F(" Z="));   Serial.print(flagZ);
  Serial.print(F(" N="));   Serial.print(flagN);
  Serial.print(F(" V="));   Serial.println(flagV);
}
