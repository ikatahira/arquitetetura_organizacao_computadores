Missão 2 · Fase 1

🧵 Dois núcleos, duas tarefas (FreeRTOS)

Criar duas tarefas FreeRTOS e fixar cada uma em um núcleo do ESP32. Ver cada uma imprimir o seu núcleo e piscar o seu LED em ritmos diferentes.

💻 Código da fase 1 paralelo1_nucleos.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  ESP32 PARALELO - FASE 1: dois nucleos, duas tarefas (FreeRTOS)

  O ESP32 tem DOIS nucleos (core 0 e core 1) e roda um sistema operacional de tempo
  real, o FreeRTOS. Aqui criamos duas TAREFAS e "fixamos" cada uma em um nucleo.
  Elas rodam AO MESMO TEMPO (paralelismo real, nao so alternancia como no Uno).

  Placa na IDE: "ESP32 Dev Module"  |  Velocidade do Monitor Serial: 115200
  Ligacoes: LED externo (300 ohms) no GPIO 4 -> GND (o GPIO 2 costuma ter LED na placa).
*/
const int PIN_LED_A = 2;     // LED da propria placa (na maioria dos modelos)
const int PIN_LED_B = 4;     // LED externo

void tarefaA(void *parametro) {
  pinMode(PIN_LED_A, OUTPUT);
  for (;;) {
    digitalWrite(PIN_LED_A, !digitalRead(PIN_LED_A));
    Serial.printf("[tarefa A] rodando no nucleo %d\n", xPortGetCoreID());
    vTaskDelay(500 / portTICK_PERIOD_MS);          // 500 ms
  }
}

void tarefaB(void *parametro) {
  pinMode(PIN_LED_B, OUTPUT);
  for (;;) {
    digitalWrite(PIN_LED_B, !digitalRead(PIN_LED_B));
    Serial.printf("[tarefa B] rodando no nucleo %d\n", xPortGetCoreID());
    vTaskDelay(1300 / portTICK_PERIOD_MS);         // 1300 ms
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("=== ESP32 PARALELO - Fase 1 ===");
  Serial.printf("setup() e loop() rodam no nucleo %d\n", xPortGetCoreID());
  Serial.printf("Frequencia da CPU: %d MHz\n", getCpuFrequencyMhz());

  // xTaskCreatePinnedToCore(funcao, nome, pilha em bytes, parametro, prioridade, handle, nucleo)
  xTaskCreatePinnedToCore(tarefaA, "tarefaA", 4096, NULL, 1, NULL, 0);   // nucleo 0
  xTaskCreatePinnedToCore(tarefaB, "tarefaB", 4096, NULL, 1, NULL, 1);   // nucleo 1
}

void loop() {
  vTaskDelay(1000 / portTICK_PERIOD_MS);
}
Missão 3 · Fase 2

📈 Speedup, eficiência e a Lei de Amdahl

Contar os primos até 200000 em 1 núcleo, em 2 núcleos com divisão em blocos e em 2 núcleos com divisão intercalada. Calcular speedup, eficiência e a fração paralelizável de Amdahl.

💻 Código da fase 2 paralelo2_speedup.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  ESP32 PARALELO - FASE 2: speedup, eficiencia e a Lei de Amdahl

  Problema: contar quantos numeros PRIMOS existem de 2 ate LIMITE.
  Medimos o tempo de 3 estrategias:
    1) UM nucleo faz tudo (tempo T1)
    2) DOIS nucleos, cada um com UMA METADE do intervalo (divisao em blocos)
    3) DOIS nucleos, alternando os numeros (par/impar -> divisao intercalada)

  Metricas:
    speedup = T1 / T2        eficiencia = speedup / numero de nucleos
  Os numeros maiores custam MAIS para testar: a divisao em blocos fica DESBALANCEADA
  (um nucleo termina antes e espera o outro). A divisao intercalada equilibra a carga.

  Placa na IDE: "ESP32 Dev Module" | Monitor Serial: 115200
*/
const uint32_t LIMITE = 200000;

volatile uint32_t contagem[2];
volatile uint32_t tempoNucleo[2];
SemaphoreHandle_t fimTarefa[2];

bool ehPrimo(uint32_t n) {
  if (n < 2) return false;
  if (n % 2 == 0) return n == 2;
  for (uint32_t d = 3; d * d <= n; d += 2) {
    if (n % d == 0) return false;
  }
  return true;
}

// ---------- 1 nucleo ----------
uint32_t contarSequencial() {
  uint32_t c = 0;
  for (uint32_t n = 2; n <= LIMITE; n++) if (ehPrimo(n)) c++;
  return c;
}

// ---------- 2 nucleos: o parametro diz o modo e o indice do nucleo ----------
struct Trabalho { uint8_t id; bool intercalado; };
Trabalho trabalhos[2];

void tarefaContar(void *p) {
  Trabalho *t = (Trabalho *)p;
  uint32_t ini = micros();
  uint32_t c = 0;
  if (t->intercalado) {
    for (uint32_t n = 2 + t->id; n <= LIMITE; n += 2) if (ehPrimo(n)) c++;
  } else {
    uint32_t meio = LIMITE / 2;
    uint32_t de = (t->id == 0) ? 2 : meio + 1;
    uint32_t ate = (t->id == 0) ? meio : LIMITE;
    for (uint32_t n = de; n <= ate; n++) if (ehPrimo(n)) c++;
  }
  contagem[t->id] = c;
  tempoNucleo[t->id] = micros() - ini;
  xSemaphoreGive(fimTarefa[t->id]);
  vTaskDelete(NULL);
}

uint32_t contarParalelo(bool intercalado) {
  for (uint8_t i = 0; i < 2; i++) { trabalhos[i].id = i; trabalhos[i].intercalado = intercalado; }
  xTaskCreatePinnedToCore(tarefaContar, "n0", 4096, &trabalhos[0], 1, NULL, 0);
  xTaskCreatePinnedToCore(tarefaContar, "n1", 4096, &trabalhos[1], 1, NULL, 1);
  xSemaphoreTake(fimTarefa[0], portMAX_DELAY);       // espera as duas terminarem (barreira)
  xSemaphoreTake(fimTarefa[1], portMAX_DELAY);
  return contagem[0] + contagem[1];
}

void executarTestes() {
  Serial.printf("Contando primos ate %u...\n", (unsigned)LIMITE);

  uint32_t t0 = micros();
  uint32_t r1 = contarSequencial();
  uint32_t T1 = micros() - t0;
  Serial.printf("1 nucleo           : %u primos em %u ms\n", (unsigned)r1, (unsigned)(T1 / 1000));

  t0 = micros();
  uint32_t r2 = contarParalelo(false);
  uint32_t T2 = micros() - t0;
  Serial.printf("2 nucleos (blocos) : %u primos em %u ms | nucleo0 %u ms, nucleo1 %u ms\n",
                (unsigned)r2, (unsigned)(T2 / 1000), (unsigned)(tempoNucleo[0] / 1000), (unsigned)(tempoNucleo[1] / 1000));
  Serial.printf("   speedup = %.2f | eficiencia = %.0f%%\n", (double)T1 / T2, 100.0 * T1 / T2 / 2);

  t0 = micros();
  uint32_t r3 = contarParalelo(true);
  uint32_t T3 = micros() - t0;
  Serial.printf("2 nucleos (alterna): %u primos em %u ms | nucleo0 %u ms, nucleo1 %u ms\n",
                (unsigned)r3, (unsigned)(T3 / 1000), (unsigned)(tempoNucleo[0] / 1000), (unsigned)(tempoNucleo[1] / 1000));
  Serial.printf("   speedup = %.2f | eficiencia = %.0f%%\n", (double)T1 / T3, 100.0 * T1 / T3 / 2);

  // Lei de Amdahl: S = 1 / ((1 - p) + p/N). Estimamos a fracao paralela p a partir do speedup medido (N = 2)
  double S = (double)T1 / T3;
  double p = (S > 1.0) ? (2.0 * (1.0 - 1.0 / S)) : 0.0;
  if (p > 1.0) p = 1.0;
  Serial.printf("Fracao paralelizavel estimada (Amdahl, N=2): p = %.2f\n", p);
  Serial.printf("Speedup maximo teorico com infinitos nucleos: %.1f\n", p < 1.0 ? 1.0 / (1.0 - p) : 999.0);
  Serial.println(r1 == r2 && r2 == r3 ? "Resultados iguais nas 3 estrategias." : "ATENCAO: resultados diferentes!");
}

void setup() {
  Serial.begin(115200);
  delay(500);
  fimTarefa[0] = xSemaphoreCreateBinary();
  fimTarefa[1] = xSemaphoreCreateBinary();
  Serial.println("=== ESP32 PARALELO - Fase 2 ===");
  Serial.println("Envie qualquer tecla para repetir o teste.");
  executarTestes();
}

void loop() {
  if (Serial.available()) {
    while (Serial.available()) Serial.read();
    executarTestes();
  }
  delay(50);
}
Missão 4 · Fase 3

🔒 Condição de corrida, mutex e fila

Provocar uma condição de corrida (dois núcleos incrementando o mesmo contador), corrigir com mutex e trocar dados entre núcleos com uma fila.

💻 Código da fase 3 paralelo3_corrida_mutex.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  ESP32 PARALELO - FASE 3: condicao de corrida, exclusao mutua (mutex) e fila

  Dois nucleos somam 1 a um CONTADOR COMPARTILHADO, 100000 vezes cada.
  O resultado correto e 200000. Testamos tres versoes:
    1) SEM protecao  -> "condicao de corrida": os dois nucleos leem, somam e gravam
                        ao mesmo tempo e PERDEM atualizacoes (resultado menor).
    2) COM MUTEX     -> so um nucleo por vez entra na SECAO CRITICA (resultado certo,
                        porem mais lento).
    3) FILA (queue)  -> um nucleo PRODUZ valores e o outro CONSOME: comunicacao entre
                        tarefas sem compartilhar variavel.

  Placa na IDE: "ESP32 Dev Module" | Monitor Serial: 115200
*/
const uint32_t VOLTAS = 100000;
volatile uint32_t contador = 0;
SemaphoreHandle_t mutex;
SemaphoreHandle_t fim[2];
QueueHandle_t fila;
bool usarMutex = false;

void tarefaSoma(void *p) {
  uint8_t id = (uint8_t)(uintptr_t)p;
  for (uint32_t i = 0; i < VOLTAS; i++) {
    if (usarMutex) xSemaphoreTake(mutex, portMAX_DELAY);
    contador = contador + 1;                  // ler -> somar -> gravar (NAO e atomico!)
    if (usarMutex) xSemaphoreGive(mutex);
  }
  xSemaphoreGive(fim[id]);
  vTaskDelete(NULL);
}

void rodarContador(bool comMutex) {
  usarMutex = comMutex;
  contador = 0;
  uint32_t t0 = micros();
  xTaskCreatePinnedToCore(tarefaSoma, "s0", 4096, (void *)0, 1, NULL, 0);
  xTaskCreatePinnedToCore(tarefaSoma, "s1", 4096, (void *)1, 1, NULL, 1);
  xSemaphoreTake(fim[0], portMAX_DELAY);
  xSemaphoreTake(fim[1], portMAX_DELAY);
  uint32_t t = micros() - t0;
  Serial.printf("%s: contador = %u (esperado %u) | perdidas = %u | tempo = %u ms\n",
                comMutex ? "COM mutex " : "SEM mutex ",
                (unsigned)contador, (unsigned)(2 * VOLTAS), (unsigned)(2 * VOLTAS - contador), (unsigned)(t / 1000));
}

// ---------- produtor/consumidor com fila ----------
void produtor(void *p) {
  for (int i = 1; i <= 5; i++) {
    xQueueSend(fila, &i, portMAX_DELAY);
    Serial.printf("[produtor, nucleo %d] enviou %d\n", xPortGetCoreID(), i);
    vTaskDelay(300 / portTICK_PERIOD_MS);
  }
  vTaskDelete(NULL);
}

void consumidor(void *p) {
  int v;
  for (int k = 0; k < 5; k++) {
    xQueueReceive(fila, &v, portMAX_DELAY);
    Serial.printf("   [consumidor, nucleo %d] recebeu %d\n", xPortGetCoreID(), v);
  }
  xSemaphoreGive(fim[0]);
  vTaskDelete(NULL);
}

void rodarFila() {
  Serial.println("--- fila: produtor (nucleo 0) -> consumidor (nucleo 1) ---");
  xTaskCreatePinnedToCore(consumidor, "cons", 4096, NULL, 1, NULL, 1);
  xTaskCreatePinnedToCore(produtor, "prod", 4096, NULL, 1, NULL, 0);
  xSemaphoreTake(fim[0], portMAX_DELAY);
}

void setup() {
  Serial.begin(115200);
  delay(500);
  mutex = xSemaphoreCreateMutex();
  fim[0] = xSemaphoreCreateBinary();
  fim[1] = xSemaphoreCreateBinary();
  fila = xQueueCreate(4, sizeof(int));
  Serial.println("=== ESP32 PARALELO - Fase 3 ===");
  rodarContador(false);
  rodarContador(false);
  rodarContador(true);
  rodarFila();
  Serial.println("Fim. Rode de novo (botao EN) e compare os resultados SEM mutex.");
}

void loop() {
  delay(1000);
}
