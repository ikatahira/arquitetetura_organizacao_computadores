/*
  RADAR BINARIO - FASE 1: Bases numericas e codificacao de dados
  Disciplina: Arquitetura e Organizacao de Computadores

  O que o programa faz:
    1. Mede a distancia com o sensor HC-SR04 (em cm).
    2. Converte a distancia em uma "faixa" de 0 a 7 (cada faixa = 5 cm).
    3. Mostra a faixa em BINARIO nos 3 LEDs:
         LED verde   = bit 0 (peso 1)  <- bit menos significativo (LSB)
         LED amarelo = bit 1 (peso 2)
         LED vermelho= bit 2 (peso 4)  <- bit mais significativo (MSB)
    4. Escreve no Monitor Serial o mesmo valor em DECIMAL, BINARIO,
       OCTAL e HEXADECIMAL, alem do codigo ASCII do digito.

  Ligacoes (confira no seu circuito):
    TRIG -> pino 8      ECHO -> pino 2
    LED verde -> 11     LED amarelo -> 12     LED vermelho -> 13
    (cada LED com resistor de 220 ou 330 ohms para o GND)
*/

const uint8_t PIN_TRIG         = 8;
const uint8_t PIN_ECHO         = 2;
const uint8_t PIN_LED_VERDE    = 11;  // bit 0
const uint8_t PIN_LED_AMARELO  = 12;  // bit 1
const uint8_t PIN_LED_VERMELHO = 13;  // bit 2
const uint8_t CM_POR_FAIXA     = 5;

// Escreve "valor" em "bits" digitos binarios (com zeros a esquerda).
void imprimirBinario(uint16_t valor, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) {
    Serial.print((valor >> i) & 1);
  }
}

// Cada LED mostra UM bit do numero.
void mostrarBits(uint8_t valor) {
  digitalWrite(PIN_LED_VERDE,    (valor >> 0) & 1);
  digitalWrite(PIN_LED_AMARELO,  (valor >> 1) & 1);
  digitalWrite(PIN_LED_VERMELHO, (valor >> 2) & 1);
}

// Dispara o sensor e devolve a distancia em cm (ou -1 se nao houver eco).
long lerDistanciaCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  unsigned long us = pulseIn(PIN_ECHO, HIGH, 30000UL);
  if (us == 0) return -1;
  return us / 58;   // ida e volta do som: ~58 us por cm
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_LED_VERDE, OUTPUT);
  pinMode(PIN_LED_AMARELO, OUTPUT);
  pinMode(PIN_LED_VERMELHO, OUTPUT);
  Serial.println(F("=== RADAR BINARIO - Fase 1 ==="));
  Serial.println(F("Aproxime e afaste a mao do sensor e observe os LEDs."));
}

void loop() {
  long cm = lerDistanciaCm();
  if (cm < 0) {
    Serial.println(F("Sem eco (objeto fora do alcance?)"));
    mostrarBits(0);
    delay(300);
    return;
  }

  uint8_t faixa = cm / CM_POR_FAIXA;
  if (faixa > 7) faixa = 7;       // so existem 3 LEDs: maximo 111b = 7
  mostrarBits(faixa);

  // --- A distancia em varias bases ---
  Serial.print(F("dist = "));
  Serial.print(cm);
  Serial.print(F(" cm | BIN8 = "));
  imprimirBinario(constrain(cm, 0, 255), 8);
  Serial.print(F(" | HEX = 0x"));
  Serial.print(cm, HEX);

  // --- A faixa (o que os LEDs mostram) em varias bases ---
  Serial.print(F(" || faixa DEC = "));
  Serial.print(faixa);
  Serial.print(F(" | BIN = "));
  imprimirBinario(faixa, 3);
  Serial.print(F(" | OCT = "));
  Serial.print(faixa, OCT);
  Serial.print(F(" | HEX = 0x"));
  Serial.print(faixa, HEX);

  // --- Codificacao de caracteres: o digito '4' NAO vale 4 ---
  Serial.print(F(" | ASCII do digito '"));
  Serial.print((char)('0' + faixa));
  Serial.print(F("' = "));
  Serial.println('0' + faixa);

  delay(400);
}

/*
  RADAR BINARIO - FASE 2: Logica digital, codigo Gray e paridade

  Novidades em relacao a Fase 1:
    - Comandos no Monitor Serial:  b = LEDs em binario  |  g = LEDs em codigo Gray
    - Codigo Gray: de um numero para o seguinte muda so 1 bit.
    - Bit de paridade par (deteccao de erro).
    - Operacoes logicas bit a bit: AND, OR, XOR, NOT.
    - Somador de 3 bits montado so com portas logicas (XOR, AND, OR).

  Ligacoes: as mesmas da Fase 1.
*/

const uint8_t PIN_TRIG         = 8;
const uint8_t PIN_ECHO         = 2;
const uint8_t PIN_LED_VERDE    = 11;  // bit 0
const uint8_t PIN_LED_AMARELO  = 12;  // bit 1
const uint8_t PIN_LED_VERMELHO = 13;  // bit 2
const uint8_t CM_POR_FAIXA     = 5;
const uint8_t MASCARA          = 0b101;   // usada nas operacoes logicas

bool modoGray = false;
uint8_t faixaAnterior = 0;

void imprimirBinario(uint16_t valor, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((valor >> i) & 1);
}

void mostrarBits(uint8_t valor) {
  digitalWrite(PIN_LED_VERDE,    (valor >> 0) & 1);
  digitalWrite(PIN_LED_AMARELO,  (valor >> 1) & 1);
  digitalWrite(PIN_LED_VERMELHO, (valor >> 2) & 1);
}

long lerDistanciaCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  unsigned long us = pulseIn(PIN_ECHO, HIGH, 30000UL);
  if (us == 0) return -1;
  return us / 58;
}

// ---------- Codigo Gray ----------
uint8_t paraGray(uint8_t b) { return b ^ (b >> 1); }

uint8_t deGray(uint8_t g) {
  uint8_t b = g;
  while (g >>= 1) b ^= g;
  return b;
}

// ---------- Paridade par: 1 se a quantidade de bits 1 for impar ----------
uint8_t paridadePar(uint8_t v) {
  uint8_t p = 0;
  while (v) {
    p ^= (v & 1);
    v >>= 1;
  }
  return p;
}

// ---------- Somador completo (1 bit) so com portas logicas ----------
void somaCompleta(uint8_t a, uint8_t b, uint8_t cin, uint8_t &soma, uint8_t &cout) {
  soma = a ^ b ^ cin;                 // duas portas XOR
  cout = (a & b) | (cin & (a ^ b));   // portas AND e OR
}

// ---------- Somador de 3 bits (ripple carry): 3 somadores completos em fila ----------
uint8_t somar3bits(uint8_t a, uint8_t b) {
  uint8_t vaiUm = 0, resultado = 0;
  for (uint8_t i = 0; i < 3; i++) {
    uint8_t s;
    somaCompleta((a >> i) & 1, (b >> i) & 1, vaiUm, s, vaiUm);
    resultado |= s << i;
  }
  resultado |= vaiUm << 3;            // o ultimo "vai um" vira o bit 3
  return resultado;
}

void imprimirTabela() {
  Serial.println(F("--- Tabela: decimal | binario | Gray | paridade ---"));
  for (uint8_t v = 0; v < 8; v++) {
    Serial.print(v);  Serial.print(F(" | "));
    imprimirBinario(v, 3); Serial.print(F(" | "));
    imprimirBinario(paraGray(v), 3); Serial.print(F(" | "));
    Serial.println(paridadePar(v));
  }
  Serial.print(F("--- Mascara = "));
  imprimirBinario(MASCARA, 3);
  Serial.println(F(" ---"));
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_LED_VERDE, OUTPUT);
  pinMode(PIN_LED_AMARELO, OUTPUT);
  pinMode(PIN_LED_VERMELHO, OUTPUT);
  Serial.println(F("=== RADAR BINARIO - Fase 2 ==="));
  Serial.println(F("Comandos: b = LEDs em binario | g = LEDs em Gray"));
  imprimirTabela();
}

void loop() {
  // --- comandos pelo Monitor Serial ---
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'b') { modoGray = false; Serial.println(F(">> LEDs em BINARIO")); }
    if (c == 'g') { modoGray = true;  Serial.println(F(">> LEDs em GRAY")); }
  }

  long cm = lerDistanciaCm();
  if (cm < 0) { mostrarBits(0); delay(300); return; }

  uint8_t faixa = cm / CM_POR_FAIXA;
  if (faixa > 7) faixa = 7;
  uint8_t gray = paraGray(faixa);
  mostrarBits(modoGray ? gray : faixa);

  Serial.print(F("faixa BIN = ")); imprimirBinario(faixa, 3);
  Serial.print(F(" | Gray = "));   imprimirBinario(gray, 3);
  Serial.print(F(" | volta p/ binario = ")); Serial.print(deGray(gray));
  Serial.print(F(" | paridade = ")); Serial.println(paridadePar(faixa));

  // --- Operacoes logicas bit a bit com a mascara ---
  Serial.print(F("   AND = ")); imprimirBinario(faixa & MASCARA, 3);
  Serial.print(F(" | OR = "));  imprimirBinario(faixa | MASCARA, 3);
  Serial.print(F(" | XOR = ")); imprimirBinario(faixa ^ MASCARA, 3);
  Serial.print(F(" | NOT = ")); imprimirBinario((~faixa) & 0b111, 3);

  // --- Soma da leitura anterior com a atual, feita com portas logicas ---
  uint8_t soma = somar3bits(faixaAnterior, faixa);
  Serial.print(F(" | ")); Serial.print(faixaAnterior);
  Serial.print(F(" + ")); Serial.print(faixa);
  Serial.print(F(" = ")); imprimirBinario(soma, 4);
  Serial.print(F(" (")); Serial.print(soma); Serial.println(F(")"));
  faixaAnterior = faixa;

  delay(500);
}

/*
  RADAR BINARIO - FASE 3: Registradores, E/S mapeada em memoria e RISC (AVR)

  O ATmega328P (Arduino Uno) e um microcontrolador RISC de 8 bits.
  Em vez de usar digitalWrite()/pulseIn(), vamos "conversar" direto com
  os REGISTRADORES de entrada e saida:
     DDRx  = Data Direction Register (1 = saida, 0 = entrada)
     PORTx = escreve nos pinos (saida)
     PINx  = le os pinos (entrada)

  Mapa dos pinos usados:
     pino 8  (TRIG)  = PB0        pino 11 (LED verde)    = PB3
     pino 12 (LED amarelo) = PB4  pino 13 (LED vermelho) = PB5
     pino 2  (ECHO)  = PD2 (entrada)

  Ligacoes: as mesmas da Fase 1.
*/

const uint8_t LEDS_MASK = 0b00111000;   // PB5 PB4 PB3

// Escreve 3 bits direto no registrador PORTB (uma unica operacao)
void mostrarBitsDireto(uint8_t v) {
  PORTB = (PORTB & ~LEDS_MASK) | ((v & 0b111) << 3);
}

// Mede o eco por POLLING (a CPU fica "perguntando" ao pino sem parar)
long medirPolling() {
  PORTB &= ~(1 << PB0);  delayMicroseconds(2);
  PORTB |=  (1 << PB0);  delayMicroseconds(10);
  PORTB &= ~(1 << PB0);

  unsigned long t0 = micros();
  while (!(PIND & (1 << PD2))) {                 // espera o eco subir
    if (micros() - t0 > 30000UL) return -1;
  }
  unsigned long ini = micros();
  while (PIND & (1 << PD2)) {                    // espera o eco descer
    if (micros() - ini > 30000UL) return -1;
  }
  return (micros() - ini) / 58;
}

void imprimirBinario(uint16_t valor, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((valor >> i) & 1);
}

// Compara a velocidade de digitalWrite() com a escrita direta no registrador
void benchmark() {
  const uint16_t N = 10000;
  unsigned long t0 = micros();
  for (uint16_t i = 0; i < N; i++) {
    digitalWrite(13, HIGH);
    digitalWrite(13, LOW);
  }
  unsigned long tDW = micros() - t0;

  t0 = micros();
  for (uint16_t i = 0; i < N; i++) {
    PORTB |=  (1 << PB5);
    PORTB &= ~(1 << PB5);
  }
  unsigned long tReg = micros() - t0;

  Serial.println(F("--- Benchmark: 20000 escritas no pino 13 ---"));
  Serial.print(F("digitalWrite : ")); Serial.print(tDW);  Serial.print(F(" us  ("));
  Serial.print(tDW / 20000.0, 3);  Serial.println(F(" us por escrita)"));
  Serial.print(F("PORTB direto : ")); Serial.print(tReg); Serial.print(F(" us  ("));
  Serial.print(tReg / 20000.0, 3); Serial.println(F(" us por escrita)"));
  Serial.print(F("PORTB direto e ")); Serial.print((float)tDW / tReg, 1);
  Serial.println(F(" vezes mais rapido"));
}

void setup() {
  Serial.begin(9600);
  DDRB |= LEDS_MASK | (1 << PB0);     // PB0, PB3, PB4, PB5 como SAIDA
  DDRD &= ~(1 << PD2);                // PD2 como ENTRADA

  Serial.println(F("=== RADAR BINARIO - Fase 3 ==="));
  Serial.print(F("Endereco de PORTB na memoria de dados: 0x"));
  Serial.println((uint16_t)&PORTB, HEX);
  Serial.print(F("Endereco de PORTB no espaco de E/S (instrucoes IN/OUT/SBI): 0x"));
  Serial.println(_SFR_IO_ADDR(PORTB), HEX);
  benchmark();
}

void loop() {
  long cm = medirPolling();
  if (cm < 0) { mostrarBitsDireto(0); delay(300); return; }

  uint8_t faixa = cm / 5;
  if (faixa > 7) faixa = 7;
  mostrarBitsDireto(faixa);

  Serial.print(F("dist = ")); Serial.print(cm);
  Serial.print(F(" cm | faixa = ")); imprimirBinario(faixa, 3);
  Serial.print(F(" | DDRB = "));  imprimirBinario(DDRB, 8);
  Serial.print(F(" | PORTB = ")); imprimirBinario(PORTB, 8);
  Serial.print(F(" | PIND = "));  imprimirBinario(PIND, 8);
  Serial.println();
  delay(400);
}
Missão 5 · Fase 4

⚡ Interrupções e mini SO

Polling (a CPU espera) contra interrupção (o sensor avisa) e um escalonador cooperativo de tarefas.

🛠️ Mão na massa 3 XP

  1. Mova o ECHO para o pino 2. Rode no modo i e depois no p.
  2. Anote o “loop() por segundo” de cada modo.
  3. Veja a coluna “execuções”: cada tarefa roda no seu período.

🎯 Desafios

💻 Código da Fase 4 fase4_interrupcoes_mini_so.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  RADAR BINARIO - FASE 4: Interrupcoes, E/S e um "mini sistema operacional"

  Duas formas de medir o eco do sensor:
    p = POLLING     : pulseIn() trava a CPU esperando o pino (bloqueante)
    i = INTERRUPCAO : o pino 2 (INT0) avisa a CPU quando o eco muda;
                      enquanto isso a CPU faz outras tarefas.

  O loop() e um escalonador cooperativo: cada tarefa tem um periodo e
  roda quando chega a sua hora (parecido com o que um SO faz).
  O relatorio mostra quantas vezes o loop() roda por segundo:
  quanto maior, mais tempo livre a CPU tem.

  Comandos no Monitor Serial:  i = interrupcao | p = polling

  ATENCAO: o ECHO precisa estar no pino 2 (INT0). O pino 3 e o outro (INT1).
*/

const uint8_t PIN_TRIG         = 8;
const uint8_t PIN_ECHO         = 2;
const uint8_t PIN_LED_VERDE    = 11;
const uint8_t PIN_LED_AMARELO  = 12;
const uint8_t PIN_LED_VERMELHO = 13;

// ---- Variaveis compartilhadas com a ISR precisam ser "volatile" ----
volatile unsigned long tInicioEco = 0;
volatile unsigned long tFimEco    = 0;
volatile bool novoEco = false;
volatile unsigned long contInterrupcoes = 0;

bool usarInterrupcao = true;
long distCm = -1;
uint8_t faixa = 0;
unsigned long voltas = 0;
unsigned long inicioJanela = 0;

// ---- ISR: Rotina de Servico de Interrupcao (curta e rapida!) ----
void isrEco() {
  contInterrupcoes++;
  if (PIND & (1 << PD2)) {
    tInicioEco = micros();          // borda de subida: eco comecou
  } else {
    tFimEco = micros();             // borda de descida: eco terminou
    novoEco = true;
  }
}

// ---- Tarefas ----
void tarefaSensor() {
  if (usarInterrupcao) {
    // So dispara o pulso; o resultado chega depois, pela interrupcao.
    digitalWrite(PIN_TRIG, LOW);  delayMicroseconds(2);
    digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(10);
    digitalWrite(PIN_TRIG, LOW);
  } else {
    // Bloqueante: a CPU fica parada ate o eco voltar (ate 30 ms!)
    digitalWrite(PIN_TRIG, LOW);  delayMicroseconds(2);
    digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(10);
    digitalWrite(PIN_TRIG, LOW);
    unsigned long us = pulseIn(PIN_ECHO, HIGH, 30000UL);
    distCm = us ? (long)(us / 58) : -1;
  }
}

void tarefaProcessa() {
  if (!usarInterrupcao) { novoEco = false; return; }
  if (!novoEco) return;
  noInterrupts();                   // copia protegida: a ISR nao pode mexer agora
  unsigned long dur = tFimEco - tInicioEco;
  novoEco = false;
  interrupts();
  distCm = dur / 58;
}

void tarefaLeds() {
  if (distCm < 0) { faixa = 0; }
  else { faixa = distCm / 5; if (faixa > 7) faixa = 7; }
  digitalWrite(PIN_LED_VERDE,    (faixa >> 0) & 1);
  digitalWrite(PIN_LED_AMARELO,  (faixa >> 1) & 1);
  digitalWrite(PIN_LED_VERMELHO, (faixa >> 2) & 1);
}

void tarefaRelatorio();   // declarada aqui, definida mais abaixo

// ---- Tabela de tarefas: ponteiros para funcoes ----
struct Tarefa {
  const char* nome;
  uint16_t periodoMs;
  unsigned long ultima;
  uint16_t execucoes;
  void (*funcao)();
};

Tarefa tarefas[] = {
  { "sensor",    60,  0, 0, tarefaSensor    },
  { "processa",   5,  0, 0, tarefaProcessa  },
  { "leds",      50,  0, 0, tarefaLeds      },
  { "relatorio", 1000, 0, 0, tarefaRelatorio }
};
const uint8_t NUM_TAREFAS = sizeof(tarefas) / sizeof(tarefas[0]);

void tarefaRelatorio() {
  unsigned long agora = millis();
  unsigned long janela = agora - inicioJanela;
  Serial.print(usarInterrupcao ? F("[INTERRUPCAO] ") : F("[POLLING] "));
  Serial.print(F("dist = ")); Serial.print(distCm);
  Serial.print(F(" cm | faixa = ")); Serial.print(faixa);
  Serial.print(F(" | loop() por segundo = ")); Serial.print(voltas * 1000UL / janela);
  Serial.print(F(" | interrupcoes = ")); Serial.println(contInterrupcoes);
  Serial.print(F("   execucoes: "));
  for (uint8_t i = 0; i < NUM_TAREFAS; i++) {
    Serial.print(tarefas[i].nome); Serial.print('='); Serial.print(tarefas[i].execucoes);
    Serial.print(' ');
    tarefas[i].execucoes = 0;
  }
  Serial.println();
  voltas = 0;
  contInterrupcoes = 0;
  inicioJanela = agora;
}

void setup() {
  Serial.begin(9600);

}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'i') { usarInterrupcao = true;  Serial.println(F(">> modo INTERRUPCAO")); }
    if (c == 'p') { usarInterrupcao = false; Serial.println(F(">> modo POLLING")); }
  }

  // Escalonador cooperativo: roda cada tarefa quando o periodo vence
  unsigned long agora = millis();
  for (uint8_t i = 0; i < NUM_TAREFAS; i++) {
    if (agora - tarefas[i].ultima >= tarefas[i].periodoMs) {
      tarefas[i].ultima = agora;
      tarefas[i].funcao();
      tarefas[i].execucoes++;
    }
  }
  voltas++;
}

/*
  RADAR BINARIO - FASE 5: Hierarquia de memoria, enderecamento e EEPROM

  Memorias do Arduino Uno (ATmega328P):
     Registradores : 32 de 8 bits (dentro da CPU, os mais rapidos)
     SRAM          : 2 KB - variaveis, pilha e heap (volatil)
     Flash         : 32 KB - guarda o PROGRAMA (nao volatil)
     EEPROM        : 1 KB - dados que sobrevivem sem energia (escrita lenta e limitada)

  Comandos no Monitor Serial:
     m = relatorio de memoria        a = modos de enderecamento
     l = latencia SRAM x Flash x EEPROM
     h = historico das ultimas leituras
     z = zerar o recorde guardado na EEPROM

  O programa guarda na EEPROM a MENOR distancia ja medida (recorde).
  Ligue e desligue a placa: o recorde continua la!
  (No simulador do Tinkercad a EEPROM nao e preservada ao reiniciar.)

  Ligacoes: as mesmas da Fase 1.
*/
#include <EEPROM.h>
#include <avr/pgmspace.h>

const uint8_t PIN_TRIG         = 8;
const uint8_t PIN_ECHO         = 2;
const uint8_t PIN_LED_VERDE    = 11;
const uint8_t PIN_LED_AMARELO  = 12;
const uint8_t PIN_LED_VERMELHO = 13;

struct Recorde {
  uint8_t  magico;     // 0xA5 = "ja existe um recorde valido"
  int16_t  menorCm;
  uint16_t leituras;
};
const uint8_t MAGICO = 0xA5;
const int ENDERECO_EEPROM = 0;
Recorde rec;

int16_t historico[8];       // buffer circular na SRAM
uint8_t indiceHist = 0;
uint8_t bufferSram[256];    // usado no teste de latencia
volatile uint8_t sumidouro; // evita que o compilador apague o teste

const char TEXTO_FLASH[] PROGMEM = "Este texto mora na Flash, junto do programa.";
char textoRam[] = "Este texto mora na SRAM e gasta memoria de dados.";

int memoriaLivre() {
  extern int __heap_start, *__brkval;
  int v;
  return (int)&v - (__brkval == 0 ? (int)&__heap_start : (int)__brkval);
}

long lerDistanciaCm() {
  digitalWrite(PIN_TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  unsigned long us = pulseIn(PIN_ECHO, HIGH, 30000UL);
  return us ? (long)(us / 58) : -1;
}

void mostrarBits(uint8_t v) {
  digitalWrite(PIN_LED_VERDE,    (v >> 0) & 1);
  digitalWrite(PIN_LED_AMARELO,  (v >> 1) & 1);
  digitalWrite(PIN_LED_VERMELHO, (v >> 2) & 1);
}

void relatorioMemoria() {
  int local = 0;
  Serial.println(F("--- RELATORIO DE MEMORIA ---"));
  Serial.print(F("SRAM: de 0x")); Serial.print(RAMSTART, HEX);
  Serial.print(F(" ate 0x"));     Serial.print(RAMEND, HEX);
  Serial.print(F(" = "));         Serial.print(RAMEND - RAMSTART + 1);
  Serial.println(F(" bytes"));
  Serial.print(F("Flash: ")); Serial.print((long)FLASHEND + 1); Serial.println(F(" bytes"));
  Serial.print(F("EEPROM: ")); Serial.print(E2END + 1);          Serial.println(F(" bytes"));
  Serial.print(F("SRAM livre agora: ")); Serial.print(memoriaLivre()); Serial.println(F(" bytes"));
  Serial.println(F("Tamanho dos tipos (bytes):"));
  Serial.print(F("  char="));   Serial.print(sizeof(char));
  Serial.print(F(" int="));     Serial.print(sizeof(int));
  Serial.print(F(" long="));    Serial.print(sizeof(long));
  Serial.print(F(" float="));   Serial.print(sizeof(float));
  Serial.print(F(" double="));  Serial.print(sizeof(double));
  Serial.print(F(" Recorde=")); Serial.println(sizeof(Recorde));
  Serial.println(F("Enderecos:"));
  Serial.print(F("  variavel global (rec)      = 0x")); Serial.println((unsigned int)&rec, HEX);
  Serial.print(F("  vetor historico[0]         = 0x")); Serial.println((unsigned int)&historico[0], HEX);
  Serial.print(F("  variavel local (na pilha)  = 0x")); Serial.println((unsigned int)&local, HEX);
  Serial.print(F("  texto na SRAM              = 0x")); Serial.println((unsigned int)textoRam, HEX);
  Serial.print(F("  texto na Flash (PROGMEM)   = 0x")); Serial.println((unsigned int)TEXTO_FLASH, HEX);
  Serial.println(F("Lendo o texto da Flash byte a byte com pgm_read_byte():"));
  for (uint8_t i = 0; i < strlen_P(TEXTO_FLASH); i++) {
    Serial.print((char)pgm_read_byte(&TEXTO_FLASH[i]));
  }
  Serial.println();
  Serial.println(textoRam);
}

void modosEnderecamento() {
  Serial.println(F("--- MODOS DE ENDERECAMENTO (visao do programador) ---"));
  uint8_t k = 42;                       // IMEDIATO: o valor 42 vai dentro da instrucao
  Serial.print(F("Imediato        : k = 42 -> ")); Serial.println(k);

  Serial.print(F("Direto          : historico[0] no endereco fixo 0x"));
  Serial.print((unsigned int)&historico[0], HEX);
  Serial.print(F(" vale ")); Serial.println(historico[0]);

  int16_t *p = &historico[0];           // ponteiro = endereco guardado em registradores
  Serial.print(F("Indireto        : p aponta para 0x")); Serial.print((unsigned int)p, HEX);
  Serial.print(F(", *p = ")); Serial.println(*p);

  Serial.print(F("Indexado        : p[3] = ")); Serial.print(p[3]);
  Serial.print(F(" (endereco 0x")); Serial.print((unsigned int)&p[3], HEX);
  Serial.println(F(" = base + 3 * 2 bytes)"));

  int16_t soma = 0;
  for (uint8_t i = 0; i < 8; i++) soma += *p++;   // pos-incremento
  Serial.print(F("Pos-incremento  : soma dos 8 valores com *p++ = ")); Serial.println(soma);
}

void testeLatencia() {
  for (uint16_t i = 0; i < 256; i++) bufferSram[i] = i * 3;
  uint8_t soma;

  soma = 0;
  unsigned long t0 = micros();
  for (uint16_t i = 0; i < 256; i++) soma += bufferSram[i];
  unsigned long tSram = micros() - t0;
  sumidouro = soma;

  soma = 0;
  t0 = micros();
  for (uint16_t i = 0; i < 256; i++) soma += pgm_read_byte((uint16_t)i);
  unsigned long tFlash = micros() - t0;
  sumidouro = soma;

  soma = 0;
  t0 = micros();
  for (uint16_t i = 0; i < 256; i++) soma += EEPROM.read(i);
  unsigned long tEeprom = micros() - t0;
  sumidouro = soma;

  Serial.println(F("--- LATENCIA: ler 256 bytes de cada memoria ---"));
  Serial.print(F("SRAM   : ")); Serial.print(tSram);   Serial.println(F(" us"));
  Serial.print(F("Flash  : ")); Serial.print(tFlash);  Serial.println(F(" us"));
  Serial.print(F("EEPROM : ")); Serial.print(tEeprom); Serial.println(F(" us"));
  Serial.println(F("Quanto mais longe da CPU, mais lenta a memoria."));
}

void mostrarHistorico() {
  Serial.print(F("Historico (cm): "));
  for (uint8_t i = 0; i < 8; i++) {
    Serial.print(historico[(indiceHist + i) % 8]);   // do mais antigo ao mais novo
    Serial.print(' ');
  }
  Serial.println();
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_LED_VERDE, OUTPUT);
  pinMode(PIN_LED_AMARELO, OUTPUT);
  pinMode(PIN_LED_VERMELHO, OUTPUT);

  EEPROM.get(ENDERECO_EEPROM, rec);
  if (rec.magico != MAGICO) {                  // primeira vez: EEPROM "limpa" (0xFF)
    rec.magico = MAGICO; rec.menorCm = 400; rec.leituras = 0;
    EEPROM.put(ENDERECO_EEPROM, rec);
  }
  Serial.println(F("=== RADAR BINARIO - Fase 5 ==="));
  Serial.println(F("Comandos: m=memoria a=enderecamento l=latencia h=historico z=zerar recorde"));
  Serial.print(F("Recorde guardado na EEPROM: ")); Serial.print(rec.menorCm);
  Serial.print(F(" cm em ")); Serial.print(rec.leituras); Serial.println(F(" leituras"));
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'm') relatorioMemoria();
    if (c == 'a') modosEnderecamento();
    if (c == 'l') testeLatencia();
    if (c == 'h') mostrarHistorico();
    if (c == 'z') {
      rec.menorCm = 400; rec.leituras = 0;
      EEPROM.put(ENDERECO_EEPROM, rec);
      Serial.println(F(">> recorde zerado"));
    }
  }

  long cm = lerDistanciaCm();
  if (cm < 0) { mostrarBits(0); delay(200); return; }

  historico[indiceHist] = cm;
  indiceHist = (indiceHist + 1) % 8;          // buffer circular

  uint8_t faixa = cm / 5;
  if (faixa > 7) faixa = 7;
  mostrarBits(faixa);

  rec.leituras++;
  if (cm > 1 && cm < rec.menorCm) {           // so grava se bater o recorde
    rec.menorCm = cm;
    EEPROM.put(ENDERECO_EEPROM, rec);         // gravar gasta vida util da EEPROM!
    Serial.print(F("NOVO RECORDE: ")); Serial.print(cm); Serial.println(F(" cm (gravado na EEPROM)"));
  }
  delay(200);
}
