💡 Telégrafo óptico: uma UART feita de luz

Um LED envia caracteres para um sensor de luz usando o mesmo quadro de uma porta serial (start, 8 bits, paridade, stop). Você mede a taxa de erro e descobre a largura de banda do canal.

Missão 1 · Montagem

🔌 Monte o circuito

Separe os materiais e faça as ligações antes de carregar o primeiro código.

🧰 Materiais

  • 1 Arduino Uno R3 + cabo USB
  • 1 protoboard 830 pontos
  • 1 LED azul ou vermelho + resistor de 300 Ω
  • 1 sensor LDR (do pacote de 5) + resistor de 10 kΩ
  • Opcional: kit emissor e receptor IR 5 mm (fototransistor)
  • 1 LED verde e 1 LED vermelho + 2 resistores de 300 Ω (indicadores)
  • Tubo de papel preto ou fita isolante (para bloquear a luz do ambiente)
  • Jumpers

⚠️ Antes de começar

  • O LDR é lento (dezenas de milissegundos): por isso o tempo de bit padrão é de 200 ms (5 bits por segundo). Com o fototransistor IR dá para ir muito mais rápido.
  • A luz do ambiente pode atrapalhar. Faça o teste num ambiente com luz estável e use o tubo.
  • No Tinkercad o LDR (fotoresistor) existe; nesse caso, mude o brilho do ambiente no próprio componente.

Missão 2 · Fase 1

📡 O canal de luz e o ADC

Medir o sensor com o LED apagado e aceso, ver o valor do ADC de 10 bits em binário e hexadecimal, descobrir a polaridade do sensor e calcular o limiar que separa 0 de 1.

💻 Código da fase 1 optico1_calibracao.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  TELEGRAFO OPTICO - FASE 1: o canal de luz e o conversor analogico-digital (ADC)

  Um LED vai transmitir e um sensor de luz vai receber, como numa fibra optica.
  Sensor: LDR (resistor que depende da luz) OU fototransistor do kit IR.
  O Arduino le o sensor com o ADC de 10 bits: valores de 0 a 1023 (0 V a 5 V).

  Nesta fase voce:
    - mede o nivel com o LED APAGADO e com o LED ACESO,
    - descobre a POLARIDADE do sensor (a luz aumenta ou diminui a leitura?),
    - calcula o LIMIAR (meio do caminho) que separa o bit 0 do bit 1.

  Ligacoes:
    LED (qualquer cor) -> pino 8, com resistor de 300 ohms para o GND
    Sensor LDR: um terminal no 5V; o outro no pino A0 E num resistor de 10 kohms ao GND
    (fototransistor IR: coletor no 5V por um resistor de 10 kohms e no pino A0;
     emissor no GND. A polaridade e detectada sozinha.)
    Aponte o LED para o sensor (um tubinho de papel preto ajuda a bloquear a luz do ambiente).
*/
const uint8_t PIN_LED = 8;
const uint8_t PIN_SENSOR = A0;

int lerMedia(uint8_t n) {
  long soma = 0;
  for (uint8_t i = 0; i < n; i++) { soma += analogRead(PIN_SENSOR); delay(5); }
  return soma / n;
}

void imprimirBinario(uint16_t v, uint8_t bits) {
  for (int8_t i = bits - 1; i >= 0; i--) Serial.print((v >> i) & 1);
}

void imprimirLeitura(const __FlashStringHelper* nome, int v) {
  Serial.print(nome);
  Serial.print(F(": ADC = ")); Serial.print(v);
  Serial.print(F(" | BIN10 = ")); imprimirBinario(v, 10);
  Serial.print(F(" | HEX = 0x")); Serial.print(v, HEX);
  Serial.print(F(" | tensao = ")); Serial.print(v * 5.0 / 1023.0, 2); Serial.println(F(" V"));
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_LED, OUTPUT);
  Serial.println(F("=== TELEGRAFO OPTICO - Fase 1: calibracao ==="));
  Serial.print(F("Resolucao do ADC: 5 V / 1023 = ")); Serial.print(5000.0 / 1023.0, 2); Serial.println(F(" mV por degrau"));
}

void loop() {
  digitalWrite(PIN_LED, LOW);  delay(400);
  int apagado = lerMedia(20);
  digitalWrite(PIN_LED, HIGH); delay(400);
  int aceso = lerMedia(20);

  imprimirLeitura(F("LED apagado"), apagado);
  imprimirLeitura(F("LED aceso   "), aceso);

  bool luzAumenta = aceso > apagado;
  int limiar = (aceso + apagado) / 2;
  int contraste = abs(aceso - apagado);

  Serial.print(F("Polaridade: ")); Serial.println(luzAumenta ? F("luz AUMENTA a leitura") : F("luz DIMINUI a leitura (invertido)"));
  Serial.print(F("Limiar = ")); Serial.print(limiar); Serial.print(F(" | contraste = ")); Serial.println(contraste);
  if (contraste < 100) Serial.println(F("ATENCAO: contraste baixo. Aproxime o LED do sensor ou reduza a luz do ambiente."));
  else Serial.println(F("Canal OK para a Fase 2."));
  Serial.println(F("-----"));
  delay(1500);
}
Missão 3 · Fase 2

📦 O quadro UART: start, dados, paridade e stop

Transmitir um caractere por luz num quadro de 11 bits e ler cada bit no meio do tempo de bit. Conferir start, paridade e stop.

💻 Código da fase 2 optico2_quadro_uart.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  TELEGRAFO OPTICO - FASE 2: um quadro (frame) estilo UART

  Digite UM caractere no Monitor Serial. O Arduino o transmite pela luz do LED
  e o recebe pelo sensor (teste "loopback"). O QUADRO tem 11 bits:

      1 bit START (0)  +  8 bits de DADOS (LSB primeiro)  +  1 bit de PARIDADE PAR  +  1 bit STOP (1)

  A luz e a "linha": LED ACESO = 1 (estado de repouso) e LED APAGADO = 0.
  A velocidade e o TEMPO DE BIT (TBIT_MS). O receptor le o sensor no MEIO de cada bit.

  Ligacoes: as mesmas da Fase 1.
*/
const uint8_t PIN_LED = 8;
const uint8_t PIN_SENSOR = A0;
const uint8_t PIN_OK = 10;      // LED verde (opcional): quadro valido
const uint8_t PIN_ERRO = 11;    // LED vermelho (opcional): erro
unsigned int TBIT_MS = 200;     // 200 ms por bit = 5 bits por segundo (LDR e lento!)

int limiar = 512;
bool luzAumenta = true;

int lerMedia(uint8_t n) {
  long soma = 0;
  for (uint8_t i = 0; i < n; i++) { soma += analogRead(PIN_SENSOR); delay(3); }
  return soma / n;
}

void calibrar() {
  digitalWrite(PIN_LED, LOW);  delay(400); int apagado = lerMedia(20);
  digitalWrite(PIN_LED, HIGH); delay(400); int aceso = lerMedia(20);
  luzAumenta = aceso > apagado;
  limiar = (aceso + apagado) / 2;
  Serial.print(F("calibrado: apagado=")); Serial.print(apagado);
  Serial.print(F(" aceso=")); Serial.print(aceso);
  Serial.print(F(" limiar=")); Serial.println(limiar);
}

bool lerBitDoSensor() {
  int v = analogRead(PIN_SENSOR);
  return luzAumenta ? (v > limiar) : (v < limiar);
}

uint8_t paridadePar(uint8_t v) {
  uint8_t p = 0;
  while (v) { p ^= (v & 1); v >>= 1; }
  return p;
}

// Transmite e, no mesmo instante, "recebe" cada bit lendo o sensor no meio do bit
void enviarEReceber(uint8_t byteEnviado, uint16_t &quadroRecebido) {
  uint16_t quadro = 0;
  quadro |= 0;                                        // bit 0: START = 0
  quadro |= (uint16_t)byteEnviado << 1;               // bits 1 a 8: dados (LSB primeiro)
  quadro |= (uint16_t)paridadePar(byteEnviado) << 9;  // bit 9: paridade par
  quadro |= (uint16_t)1 << 10;                        // bit 10: STOP = 1

  quadroRecebido = 0;
  for (uint8_t i = 0; i < 11; i++) {
    digitalWrite(PIN_LED, (quadro >> i) & 1);
    delay(TBIT_MS / 2);
    if (lerBitDoSensor()) quadroRecebido |= (uint16_t)1 << i;   // amostra no MEIO do bit
    delay(TBIT_MS - TBIT_MS / 2);
  }
  digitalWrite(PIN_LED, HIGH);                        // linha volta ao repouso (1)
}

void imprimirQuadro(uint16_t q) {
  Serial.print(F("START="));  Serial.print(q & 1);
  Serial.print(F(" DADOS=")); for (int8_t i = 8; i >= 1; i--) Serial.print((q >> i) & 1);
  Serial.print(F(" PAR="));   Serial.print((q >> 9) & 1);
  Serial.print(F(" STOP="));  Serial.print((q >> 10) & 1);
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_LED, OUTPUT); pinMode(PIN_OK, OUTPUT); pinMode(PIN_ERRO, OUTPUT);
  Serial.println(F("=== TELEGRAFO OPTICO - Fase 2 ==="));
  calibrar();
  Serial.print(F("Tempo de bit = ")); Serial.print(TBIT_MS); Serial.print(F(" ms -> ")); Serial.print(1000.0 / TBIT_MS, 1); Serial.println(F(" bits/s"));
  Serial.println(F("Digite um caractere e envie."));
}

void loop() {
  if (!Serial.available()) return;
  char c = Serial.read();
  if (c == '\n' || c == '\r') return;

  uint16_t rx;
  enviarEReceber((uint8_t)c, rx);

  uint8_t dados = (rx >> 1) & 0xFF;
  bool startOk = (rx & 1) == 0;
  bool stopOk  = ((rx >> 10) & 1) == 1;
  bool parOk   = ((rx >> 9) & 1) == paridadePar(dados);

  Serial.print(F("enviado '")); Serial.print(c); Serial.print(F("' = 0x")); Serial.print((uint8_t)c, HEX);
  Serial.print(F(" | quadro RX: ")); imprimirQuadro(rx); Serial.println();
  Serial.print(F("recebido '")); Serial.print((char)dados); Serial.print(F("' = 0x")); Serial.print(dados, HEX);
  Serial.print(F(" | start ")); Serial.print(startOk ? F("ok") : F("ERRO"));
  Serial.print(F(" | paridade ")); Serial.print(parOk ? F("ok") : F("ERRO"));
  Serial.print(F(" | stop ")); Serial.println(stopOk ? F("ok") : F("ERRO"));

  bool tudoOk = startOk && stopOk && parOk && (dados == (uint8_t)c);
  digitalWrite(PIN_OK, tudoOk);
  digitalWrite(PIN_ERRO, !tudoOk);
}
Missão 4 · Fase 3

📉 Texto, taxa de erro (BER) e largura de banda

Enviar frases inteiras, medir a taxa de erro de bits (BER) e comparar a taxa bruta com a taxa útil. Diminuir o tempo de bit até o canal começar a errar.

💻 Código da fase 3 optico3_texto_ber.ino
Cole no Tinkercad (Código → Texto) ou na IDE do Arduino.
/*
  TELEGRAFO OPTICO - FASE 3: textos, taxa de erro (BER) e largura de banda

  Digite uma frase no Monitor Serial: cada caractere vira um quadro de 11 bits.
  No fim o programa mostra:
    - quantos caracteres chegaram certos e quantos tiveram erro de paridade/dados
    - a TAXA DE ERRO DE BITS (BER) = bits errados / bits enviados
    - a taxa BRUTA (bits/s no canal) e a taxa UTIL (bits de dados por segundo)

  Para mudar a velocidade, envie uma linha que comeca com # e o tempo de bit em ms:
      #100   (100 ms por bit)     #20   (20 ms por bit)
  Descubra a velocidade em que o seu sensor comeca a errar!

  Ligacoes: as mesmas da Fase 1.
*/
const uint8_t PIN_LED = 8;
const uint8_t PIN_SENSOR = A0;
const uint8_t PIN_OK = 10;
const uint8_t PIN_ERRO = 11;
unsigned int TBIT_MS = 200;

int limiar = 512;
bool luzAumenta = true;

int lerMedia(uint8_t n) {
  long soma = 0;
  for (uint8_t i = 0; i < n; i++) { soma += analogRead(PIN_SENSOR); delay(3); }
  return soma / n;
}

void calibrar() {
  digitalWrite(PIN_LED, LOW);  delay(400); int apagado = lerMedia(20);
  digitalWrite(PIN_LED, HIGH); delay(400); int aceso = lerMedia(20);
  luzAumenta = aceso > apagado;
  limiar = (aceso + apagado) / 2;
  Serial.print(F("calibrado: limiar=")); Serial.print(limiar);
  Serial.print(F(" polaridade=")); Serial.println(luzAumenta ? F("normal") : F("invertida"));
}

bool lerBitDoSensor() {
  int v = analogRead(PIN_SENSOR);
  return luzAumenta ? (v > limiar) : (v < limiar);
}

uint8_t paridadePar(uint8_t v) {
  uint8_t p = 0;
  while (v) { p ^= (v & 1); v >>= 1; }
  return p;
}

uint8_t contarBits(uint16_t v) {
  uint8_t n = 0;
  while (v) { n += v & 1; v >>= 1; }
  return n;
}

uint16_t montarQuadro(uint8_t b) {
  return ((uint16_t)1 << 10) | ((uint16_t)paridadePar(b) << 9) | ((uint16_t)b << 1);
}

uint16_t enviarEReceber(uint16_t quadro) {
  uint16_t rx = 0;
  for (uint8_t i = 0; i < 11; i++) {
    digitalWrite(PIN_LED, (quadro >> i) & 1);
    delay(TBIT_MS / 2);
    if (lerBitDoSensor()) rx |= (uint16_t)1 << i;
    delay(TBIT_MS - TBIT_MS / 2);
  }
  digitalWrite(PIN_LED, HIGH);
  return rx;
}

void transmitirTexto(String s) {
  unsigned int n = s.length();
  unsigned int certos = 0, errosParidade = 0, errosDados = 0, bitsErrados = 0;
  String recebido = "";
  unsigned long t0 = millis();

  for (unsigned int k = 0; k < n; k++) {
    uint8_t c = s.charAt(k);
    uint16_t tx = montarQuadro(c);
    uint16_t rx = enviarEReceber(tx);
    uint8_t dados = (rx >> 1) & 0xFF;
    bool parOk = ((rx >> 9) & 1) == paridadePar(dados);
    bitsErrados += contarBits((tx ^ rx) & 0x7FF);      // XOR mostra exatamente quais bits mudaram
    if (!parOk) errosParidade++;
    if (dados != c) errosDados++;
    if (parOk && dados == c) certos++;
    recebido += (char)dados;
  }
  unsigned long ms = millis() - t0;

  Serial.print(F("Enviado : ")); Serial.println(s);
  Serial.print(F("Recebido: ")); Serial.println(recebido);
  Serial.print(F("caracteres certos = ")); Serial.print(certos); Serial.print('/'); Serial.println(n);
  Serial.print(F("erros de paridade = ")); Serial.print(errosParidade);
  Serial.print(F(" | erros de dados = ")); Serial.println(errosDados);
  float ber = (float)bitsErrados / (n * 11.0);
  Serial.print(F("BER = ")); Serial.print(bitsErrados); Serial.print(F(" bits errados de ")); Serial.print(n * 11);
  Serial.print(F(" = ")); Serial.println(ber, 4);
  Serial.print(F("taxa bruta = ")); Serial.print(1000.0 / TBIT_MS, 1); Serial.print(F(" bits/s | taxa util = "));
  Serial.print(n * 8000.0 / ms, 1); Serial.println(F(" bits de dados/s"));
  digitalWrite(PIN_OK, certos == n);
  digitalWrite(PIN_ERRO, certos != n);
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_LED, OUTPUT); pinMode(PIN_OK, OUTPUT); pinMode(PIN_ERRO, OUTPUT);
  Serial.println(F("=== TELEGRAFO OPTICO - Fase 3 ==="));
  calibrar();
  Serial.println(F("Digite uma frase. Para mudar a velocidade: #<ms por bit>"));
}

void loop() {
  if (!Serial.available()) return;
  String linha = Serial.readStringUntil('\n');
  linha.trim();
  if (linha.length() == 0) return;
  if (linha.charAt(0) == '#') {
    int v = linha.substring(1).toInt();
    if (v >= 2 && v <= 1000) { TBIT_MS = v; Serial.print(F(">> tempo de bit = ")); Serial.print(TBIT_MS); Serial.println(F(" ms")); }
    return;
  }
  if (linha.length() > 40) linha = linha.substring(0, 40);
  transmitirTexto(linha);
}
